12/// Supported WebSocket commands.
13#[derive(Debug, Clone, Deserialize)]
14#[serde(tag = "type", rename_all = "snake_case")]
15pub enum WsCommand {
16    /// Scan for devices.
17    Scan {
18        /// Scan duration in seconds (1–600).
19        duration_secs: ScanDuration,
20    },
21    /// Connect to a device.
22    Connect {
23        /// Device MAC address.
24        address: MacAddress,
25    },
26    /// Disconnect from a device.
27    Disconnect {
28        /// Device MAC address.
29        address: MacAddress,
30    },
31    /// Synchronize device time to system time.
32    SyncTime {
33        /// Device MAC address.
34        address: MacAddress,
35    },
36    /// Read all alarms.
37    ReadAlarms {
38        /// Device MAC address.
39        address: MacAddress,
40    },
41    /// Set an alarm.
42    SetAlarm {
43        /// Device MAC address.
44        address: MacAddress,
45        /// Slot index (0–15).
46        slot: AlarmSlotIndex,
47        /// Alarm time (hour and minute).
48        time: ClockTime,
49        /// Repeat day bitmask.
50        repeat_mask: DayMask,
51        /// Whether the alarm is enabled.
52        enabled: bool,
53        /// Whether snooze is enabled.
54        snooze: bool,
55    },
56    /// Delete an alarm.
57    DeleteAlarm {
58        /// Device MAC address.
59        address: MacAddress,
60        /// Slot index (0–15).
61        slot: AlarmSlotIndex,
62    },
63    /// Read device settings.
64    ReadSettings {
65        /// Device MAC address.
66        address: MacAddress,
67    },
68    /// Write device settings.
69    WriteSettings {
70        /// Device MAC address.
71        address: MacAddress,
72        /// Settings to write.
73        settings: DeviceSettings,
74    },
75    /// Set brightness preview.
76    SetBrightness {
77        /// Device MAC address.
78        address: MacAddress,
79        /// Brightness value.
80        value: Brightness,
81    },
82    /// Preview ringtone.
83    PreviewRingtone {
84        /// Device MAC address.
85        address: MacAddress,
86        /// Optional volume level.
87        volume: Option<Volume>,
88    },
89    /// Read firmware version.
90    ReadFirmware {
91        /// Device MAC address.
92        address: MacAddress,
93    },
94    /// Read battery level.
95    ReadBattery {
96        /// Device MAC address.
97        address: MacAddress,
98    },
99    /// Subscribe to sensor events.
100    SubscribeEvents {
101        /// Device MAC address.
102        address: MacAddress,
103    },
104}
