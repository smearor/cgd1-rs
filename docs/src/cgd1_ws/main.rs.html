12/// Command-line arguments for the WebSocket server.
13#[derive(Parser, Debug)]
14#[command(name = "cgd1-ws", about = "WebSocket and REST server for the Qingping CGD1 alarm clock")]
15struct Cli {
16    /// Bind address.
17    #[arg(long, default_value = "0.0.0.0")]
18    address: String,
19
20    /// Bind port.
21    #[arg(long, default_value_t = 3000)]
22    port: u16,
23
24    /// Verbosity level (0=warn, 1=info, 2=debug, 3=trace).
25    #[arg(short, long, default_value_t = 0)]
26    verbose: u8,
27}
28
29#[tokio::main]
30async fn main() -> Result<(), Box<dyn std::error::Error>> {
31    let cli = Cli::parse();
32
33    let filter = match cli.verbose {
34        0 => tracing_subscriber::EnvFilter::new("warn"),
35        1 => tracing_subscriber::EnvFilter::new("info"),
36        2 => tracing_subscriber::EnvFilter::new("debug"),
37        _ => tracing_subscriber::EnvFilter::new("trace"),
38    };
39    tracing_subscriber::fmt().with_env_filter(filter).init();
40
41    let state = ServerState::new().await?;
42    let router = routes::build_router(state);
43
44    let bind_addr = format!("{}:{}", cli.address, cli.port);
45    let listener = tokio::net::TcpListener::bind(&bind_addr).await?;
46    tracing::info!(addr = %bind_addr, "WebSocket server listening");
47
48    axum::serve(listener, router).await?;
49
50    Ok(())
51}
