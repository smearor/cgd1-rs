11/// Computed next alarm time with a human-readable label.
12pub struct NextAlarm {
13    /// The date-time when the alarm will fire.
14    pub when: DateTime<Local>,
15    /// Short label for display, e.g. "07:30 Mon".
16    pub label: String,
17}

19/// Compute the next upcoming alarm from a list of alarm slots.
20///
21/// Only enabled alarms are considered. Returns `None` if no enabled alarm exists.
22pub fn next_alarm(slots: &[AlarmSlot]) -> Option<NextAlarm> {
23    let now = Local::now();
24    let mut best: Option<NextAlarm> = None;
25
26    for slot in slots.iter().filter(|s| s.entry.enabled()) {
27        let hour = slot.entry.hour();
28        let minute = slot.entry.minute();
29        let mask = slot.entry.repeat_mask();
30
31        let alarm_time = NaiveTime::from_hms_opt(hour as u32, minute as u32, 0)?;
32        let when = compute_next_fire(now, alarm_time, mask)?;
33        let candidate = NextAlarm {
34            when,
35            label: format_label(when, slot),
36        };
37
38        if best.as_ref().map_or(true, |b| when < b.when) {
39            best = Some(candidate);
40        }
41    }
42
43    best
44}

46/// Compute the next date-time this alarm will fire, given the current time and repeat mask.
47fn compute_next_fire(now: DateTime<Local>, alarm_time: NaiveTime, mask: DayMask) -> Option<DateTime<Local>> {
48    let mask_val = mask.value();
49
50    // One-shot alarm: fires today if time hasn't passed, otherwise tomorrow.
51    if mask_val == DayMask::ONCE.value() {
52        let today_alarm = now.date_naive().and_time(alarm_time);
53        let today_local = Local.from_local_datetime(&today_alarm).single()?;
54        if today_local > now {
55            return Some(today_local);
56        }
57        let tomorrow = today_local + chrono::Duration::days(1);
58        return Some(tomorrow);
59    }
60
61    // Repeating alarm: find the next matching weekday.
62    // Bit 0 = Monday, ..., Bit 6 = Sunday.
63    // chrono::Weekday: Mon=0, Tue=1, ..., Sun=6 — matches our bit layout.
64    for offset in 0..7 {
65        let candidate_day = now + chrono::Duration::days(offset);
66        let weekday_bit = 1u8 << weekday_num(candidate_day.weekday());
67        if mask_val & weekday_bit == 0 {
68            continue;
69        }
70        let candidate_naive = candidate_day.date_naive().and_time(alarm_time);
71        let candidate = Local.from_local_datetime(&candidate_naive).single()?;
72        if candidate > now {
73            return Some(candidate);
74        }
75    }
76
77    // All matching days today have passed; look at next week.
78    for offset in 7..14 {
79        let candidate_day = now + chrono::Duration::days(offset);
80        let weekday_bit = 1u8 << weekday_num(candidate_day.weekday());
81        if mask_val & weekday_bit == 0 {
82            continue;
83        }
84        let candidate_naive = candidate_day.date_naive().and_time(alarm_time);
85        let candidate = Local.from_local_datetime(&candidate_naive).single()?;
86        return Some(candidate);
87    }
88
89    None
90}

92/// Format a short label for the next alarm, e.g. "07:30 Mon".
93fn format_label(when: DateTime<Local>, slot: &AlarmSlot) -> String {
94    let mask_val = slot.entry.repeat_mask().value();
95    let time_str = format!("{:02}:{:02}", slot.entry.hour(), slot.entry.minute());
96
97    if mask_val == DayMask::ONCE.value() {
98        let day_name = weekday_short(when.weekday());
99        format!("{time_str} {day_name}")
100    } else if mask_val == DayMask::EVERY_DAY.value() {
101        time_str
102    } else {
103        let day_name = weekday_short(when.weekday());
104        format!("{time_str} {day_name}")
105    }
106}

108/// Short weekday name for display.
109fn weekday_short(wd: chrono::Weekday) -> String {
110    use chrono::Weekday::*;
111    let key = match wd {
112        Mon => "weekday-mon",
113        Tue => "weekday-tue",
114        Wed => "weekday-wed",
115        Thu => "weekday-thu",
116        Fri => "weekday-fri",
117        Sat => "weekday-sat",
118        Sun => "weekday-sun",
119    };
120    get(key)
121}
122
123/// Convert a `Weekday` to a 0-based number (Mon=0 ... Sun=6), matching `DayMask` bit layout.
124fn weekday_num(wd: chrono::Weekday) -> u8 {
125    use chrono::Weekday::*;
126    match wd {
127        Mon => 0,
128        Tue => 1,
129        Wed => 2,
130        Thu => 3,
131        Fri => 4,
132        Sat => 5,
133        Sun => 6,
134    }
135}
