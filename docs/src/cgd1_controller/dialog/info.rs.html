1use gtk4::Align;
2use gtk4::Box;
3use gtk4::Button;
4use gtk4::Label;
5use gtk4::Orientation;
6use gtk4::Picture;
7use gtk4::Window;
8use gtk4::glib;
9use gtk4::prelude::*;
10
11/// SVG image of the CGD1 alarm clock rendered as a data URI.
12const CLOCK_SVG: &str = include_str!("../../assets/clock.svg");
13
14/// Info dialog showing application title, clock image, repository link, and license.
15#[allow(dead_code)]
16pub struct InfoDialog {
17    window: Window,
18}
19
20impl InfoDialog {
21    /// Create and show the info dialog.
22    pub fn new(parent: &Window) -> Self {
23        let window = gtk4::Window::builder()
24            .title("Info — Alarm Clock CGD1")
25            .transient_for(parent)
26            .modal(true)
27            .default_width(360)
28            .default_height(420)
29            .build();
30
31        let main_box = Box::builder()
32            .orientation(Orientation::Vertical)
33            .spacing(16)
34            .margin_top(20)
35            .margin_bottom(20)
36            .margin_start(20)
37            .margin_end(20)
38            .halign(Align::Center)
39            .build();
40
41        let title = Label::builder()
42            .label("Alarm Clock CGD1")
43            .css_classes(["title-1"])
44            .halign(Align::Center)
45            .build();
46        main_box.append(&title);
47
48        let svg_bytes = glib::Bytes::from(CLOCK_SVG.as_bytes());
49        let texture = gtk4::gdk::Texture::from_bytes(&svg_bytes).ok();
50        if let Some(texture) = texture {
51            let picture = Picture::builder().paintable(&texture).halign(Align::Center).build();
52            main_box.append(&picture);
53        }
54
55        let link_button = gtk4::LinkButton::builder()
56            .label("github.com/smearor/cgd1-rs")
57            .uri("https://github.com/smearor/cgd1-rs")
58            .halign(Align::Center)
59            .build();
60        main_box.append(&link_button);
61
62        let license_label = Label::builder()
63            .label("Licensed under the MIT License\n\nCopyright (c) 2024 smearor\n\nPermission is hereby granted, free of charge, to any person obtaining a copy of this software and associated documentation files (the \"Software\"), to deal in the Software without restriction, including without limitation the rights to use, copy, modify, merge, publish, distribute, sublicense, and/or sell copies of the Software, and to permit persons to whom the Software is furnished to do so, subject to the following conditions:\n\nThe above copyright notice and this permission notice shall be included in all copies or substantial portions of the Software.\n\nTHE SOFTWARE IS PROVIDED \"AS IS\", WITHOUT WARRANTY OF ANY KIND.")
64            .wrap(true)
65            .halign(Align::Center)
66            .valign(Align::Start)
67            .vexpand(true)
68            .build();
69        main_box.append(&license_label);
70
71        let close_button = Button::builder().label("Close").halign(Align::Center).build();
72        let win = window.clone();
73        close_button.connect_clicked(move |_| {
74            win.close();
75        });
76        main_box.append(&close_button);
77
78        window.set_child(Some(&main_box));
79        window.present();
80
81        Self { window }
82    }
83}
