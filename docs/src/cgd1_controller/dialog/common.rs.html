8/// Create a labeled settings row with a fixed-width label.
9pub fn settings_row(label_text: &str) -> Box {
10    let row = Box::builder()
11        .orientation(Orientation::Horizontal)
12        .spacing(12)
13        .margin_top(4)
14        .margin_bottom(4)
15        .build();
16    let label = Label::builder()
17        .label(label_text)
18        .xalign(1.0f32)
19        .width_chars(18)
20        .css_classes(["settings-label"])
21        .build();
22    row.append(&label);
23    row
24}
25
26/// Create a frame with an icon + title header.
27pub fn settings_frame(icon_name: &str, title: &str) -> Frame {
28    let header = Box::builder().orientation(Orientation::Horizontal).spacing(6).build();
29    let icon = Image::builder().icon_name(icon_name).css_classes(["frame-icon"]).build();
30    let label = Label::builder().label(title).css_classes(["frame-title"]).build();
31    header.append(&icon);
32    header.append(&label);
33
34    Frame::builder()
35        .label_widget(&header)
36        .margin_top(4)
37        .margin_bottom(4)
38        .css_classes(["settings-frame"])
39        .build()
40}
