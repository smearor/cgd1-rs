1use gtk4::Align;
2use gtk4::Box;
3use gtk4::Button;
4use gtk4::FileChooserAction;
5use gtk4::FileChooserDialog;
6use gtk4::Label;
7use gtk4::Orientation;
8use gtk4::ProgressBar;
9use gtk4::Window;
10use gtk4::prelude::*;
11
12/// Audio dialog for ringtone upload and preview.
13#[allow(dead_code)]
14pub struct AudioDialog {
15    window: Window,
16}
17
18impl AudioDialog {
19    /// Create and show the audio dialog.
20    pub fn new(parent: &Window) -> Self {
21        let window = gtk4::Window::builder()
22            .title("Audio — Alarm Clock CGD1")
23            .transient_for(parent)
24            .modal(true)
25            .default_width(400)
26            .default_height(300)
27            .build();
28
29        let main_box = Box::builder()
30            .orientation(Orientation::Vertical)
31            .spacing(12)
32            .margin_top(16)
33            .margin_bottom(16)
34            .margin_start(16)
35            .margin_end(16)
36            .build();
37
38        let title = Label::builder().label("Custom Ringtone").css_classes(["title-2"]).halign(Align::Start).build();
39        main_box.append(&title);
40
41        let info_label = Label::builder()
42            .label("Upload a custom ringtone (8-bit PCM, 8 kHz, mono, max ~12 seconds).\nConnect to a device to upload.")
43            .wrap(true)
44            .halign(Align::Start)
45            .css_classes(["dim-label"])
46            .build();
47        main_box.append(&info_label);
48
49        let file_label = Label::builder().label("No file selected").halign(Align::Start).build();
50        main_box.append(&file_label);
51
52        let select_button = Button::builder().label("Select Audio File…").halign(Align::Start).build();
53        main_box.append(&select_button);
54
55        let progress = ProgressBar::builder().fraction(0.0).hexpand(true).build();
56        main_box.append(&progress);
57
58        let status_label = Label::builder().label("").halign(Align::Start).build();
59        main_box.append(&status_label);
60
61        let button_box = Box::builder().orientation(Orientation::Horizontal).spacing(8).halign(Align::End).build();
62        let preview_button = Button::builder().label("Preview Ringtone").build();
63        let upload_button = Button::builder().label("Upload").css_classes(["suggested-action"]).build();
64        let close_button = Button::builder().label("Close").build();
65        button_box.append(&preview_button);
66        button_box.append(&upload_button);
67        button_box.append(&close_button);
68        main_box.append(&button_box);
69
70        let win = window.clone();
71        close_button.connect_clicked(move |_| {
72            win.close();
73        });
74
75        let file_label_clone = file_label.clone();
76        select_button.connect_clicked(move |_| {
77            let filter = gtk4::FileFilter::new();
78            filter.set_name(Some("Audio files"));
79            filter.add_pattern("*.wav");
80            filter.add_pattern("*.raw");
81            filter.add_pattern("*.pcm");
82
83            let dialog = FileChooserDialog::builder().title("Select Audio File").action(FileChooserAction::Open).build();
84            dialog.add_filter(&filter);
85            dialog.add_buttons(&[("Cancel", gtk4::ResponseType::Cancel), ("Open", gtk4::ResponseType::Accept)]);
86
87            let label = file_label_clone.clone();
88            dialog.connect_response(move |d, response| {
89                if response == gtk4::ResponseType::Accept
90                    && let Some(file) = d.file()
91                {
92                    let path = file.path().map(|p| p.to_string_lossy().to_string()).unwrap_or_default();
93                    label.set_label(&path);
94                }
95                d.close();
96            });
97            dialog.present();
98        });
99
100        window.set_child(Some(&main_box));
101        window.present();
102
103        Self { window }
104    }
105}
