32/// Reusable alarm editor widget - can be embedded in the main window or a dialog.
33#[allow(dead_code)]
34pub struct AlarmEditorWidget {
35    /// The container box holding all alarm editor content.
36    pub container: Box,
37    /// Status label for operation feedback.
38    pub status_label: Label,
39    /// Refresh button.
40    pub refresh_button: Button,
41    /// Row widgets for reading alarm state back from device.
42    row_widgets: Vec<(TimeEntry, Switch, ToggleButton, ToggleButton, Vec<ToggleButton>)>,
43    /// Clock manager for device communication.
44    manager: Arc<ClockManager>,
45    /// Async runtime.
46    runtime: Arc<Runtime>,
47    /// Connected device address.
48    connected_address: Arc<Mutex<Option<MacAddress>>>,
49}

51/// Widgets for a single alarm row.
52#[derive(Clone)]
53struct AlarmRowWidgets {
54    time_entry: TimeEntry,
55    enabled_switch: Switch,
56    snooze_toggle: ToggleButton,
57    once_toggle: ToggleButton,
58    day_toggles: Vec<ToggleButton>,
59    set_button: Button,
60    delete_button: Button,
61}

64    /// Build the alarm editor content (without window chrome).
65    pub fn new(manager: Arc<ClockManager>, runtime: Arc<Runtime>, connected_address: Arc<Mutex<Option<MacAddress>>>, on_changed: impl Fn() + 'static) -> Self {
66        let on_changed = Arc::new(on_changed);
67        let container = Box::builder()
68            .orientation(Orientation::Vertical)
69            .spacing(8)
70            .margin_top(8)
71            .margin_bottom(8)
72            .margin_start(8)
73            .margin_end(8)
74            .build();
75
76        let scrolled = ScrolledWindow::builder()
77            .hscrollbar_policy(PolicyType::Never)
78            .vscrollbar_policy(PolicyType::Automatic)
79            .vexpand(true)
80            .max_content_height(300)
81            .propagate_natural_height(true)
82            .build();
83
84        let alarm_list_box = Box::builder().orientation(Orientation::Vertical).spacing(4).build();
85
86        let mut rows: Vec<(u8, AlarmRowWidgets)> = Vec::new();
87        for slot in 0..16u8 {
88            let (row, widgets) = create_alarm_row();
89            alarm_list_box.append(&row);
90            rows.push((slot, widgets));
91        }
92
93        scrolled.set_child(Some(&alarm_list_box));
94        container.append(&scrolled);
95
96        container.append(&Separator::new(Orientation::Horizontal));
97
98        let status_label = Label::builder().label("").css_classes(["dim-label"]).halign(Align::Start).hexpand(true).build();
99
100        let button_box = Box::builder().orientation(Orientation::Horizontal).spacing(8).halign(Align::Fill).build();
101        let refresh_button = Button::builder().label(&fl!("alarm-read-from-device")).build();
102        button_box.append(&status_label);
103        button_box.append(&refresh_button);
104        container.append(&button_box);
105
106        // Read from Device - connect button after widget construction via stored fields
107        // (handled below after struct is built)
108
109        // Set and Delete per row
110        for (slot, widgets) in rows.iter().map(|(s, w)| (*s, w.clone())) {
111            let Ok(slot_idx) = AlarmSlotIndex::new(slot) else {
112                error!(slot, "invalid alarm slot index, skipping row");
113                continue;
114            };
115            let manager_set = manager.clone();
116            let runtime_set = runtime.clone();
117            let connected_address_set = connected_address.clone();
118            let status_label_set = status_label.clone();
119            let on_changed_set = on_changed.clone();
120            let time_entry = widgets.time_entry.clone();
121            let enabled_switch = widgets.enabled_switch.clone();
122            let snooze_toggle = widgets.snooze_toggle.clone();
123            let once_toggle = widgets.once_toggle.clone();
124            let day_toggles = widgets.day_toggles.clone();
125
126            widgets.set_button.connect_clicked(move |_| {
127                let addr = *connected_address_set.lock().unwrap_or_else(|p| {
128                    warn!("mutex poisoned - recovering");
129                    p.into_inner()
130                });
131                let Some(addr) = addr else {
132                    status_label_set.set_label(&fl!("status-no-device-connected"));
133                    return;
134                };
135                let Some((hour, minute)) = time_entry.get_time() else {
136                    status_label_set.set_label(&fl!("status-invalid-time"));
137                    return;
138                };
139                let enabled = enabled_switch.is_active();
140                let snooze = snooze_toggle.is_active();
141                let repeat_mask = if once_toggle.is_active() {
142                    DayMask::ONCE
143                } else {
144                    let mut mask = 0u8;
145                    for (j, btn) in day_toggles.iter().enumerate() {
146                        if btn.is_active() {
147                            mask |= 1 << j;
148                        }
149                    }
150                    DayMask::new(mask)
151                };
152                let time = match ClockTime::new(hour, minute) {
153                    Ok(t) => t,
154                    Err(e) => {
155                        status_label_set.set_label(&fl!("status-invalid-time-error", error = e.to_string()));
156                        return;
157                    }
158                };
159                let entry = AlarmEntry::new(time, repeat_mask, enabled, snooze);
160                status_label_set.set_label(&fl!("status-setting-alarm", slot = (slot as i64)));
161                let manager_set = manager_set.clone();
162                let (tx, rx) = std::sync::mpsc::channel::<Result<(), String>>();
163                runtime_set.spawn(async move {
164                    let result = async {
165                        let device = manager_set.device(&addr).await.ok_or_else(|| ClockError::Parse("device not found".into()))?;
166                        device.set_alarm(&entry, slot_idx).await
167                    }
168                    .await;
169                    let _ = tx.send(result.map_err(|e| e.to_string()));
170                });
171                let rx = std::cell::RefCell::new(rx);
172                let status_label_set = status_label_set.clone();
173                let on_changed = on_changed_set.clone();
174                glib::source::idle_add_local(move || match rx.borrow_mut().try_recv() {
175                    Ok(result) => {
176                        match result {
177                            Ok(()) => {
178                                status_label_set.set_label(&fl!("status-alarm-set", slot = (slot as i64)));
179                                (*on_changed)();
180                            }
181                            Err(e) => status_label_set.set_label(&fl!("status-set-failed", error = e.to_string())),
182                        }
183                        ControlFlow::Break
184                    }
185                    Err(TryRecvError::Empty) => ControlFlow::Continue,
186                    Err(TryRecvError::Disconnected) => {
187                        status_label_set.set_label(&fl!("status-set-task-failed"));
188                        ControlFlow::Break
189                    }
190                });
191            });
192
193            let manager_del = manager.clone();
194            let runtime_del = runtime.clone();
195            let connected_address_del = connected_address.clone();
196            let status_label_del = status_label.clone();
197            let on_changed_del = on_changed.clone();
198
199            widgets.delete_button.connect_clicked(move |_| {
200                let addr = *connected_address_del.lock().unwrap_or_else(|p| {
201                    warn!("mutex poisoned - recovering");
202                    p.into_inner()
203                });
204                let Some(addr) = addr else {
205                    status_label_del.set_label(&fl!("status-no-device-connected"));
206                    return;
207                };
208                status_label_del.set_label(&fl!("status-deleting-alarm", slot = (slot as i64)));
209                let manager_del = manager_del.clone();
210                let (tx, rx) = std::sync::mpsc::channel::<Result<(), String>>();
211                runtime_del.spawn(async move {
212                    let result = async {
213                        let device = manager_del.device(&addr).await.ok_or_else(|| ClockError::Parse("device not found".into()))?;
214                        device.delete_alarm(slot_idx).await
215                    }
216                    .await;
217                    let _ = tx.send(result.map_err(|e| e.to_string()));
218                });
219                let rx = std::cell::RefCell::new(rx);
220                let status_label_del = status_label_del.clone();
221                let on_changed = on_changed_del.clone();
222                glib::source::idle_add_local(move || match rx.borrow_mut().try_recv() {
223                    Ok(result) => {
224                        match result {
225                            Ok(()) => {
226                                status_label_del.set_label(&fl!("status-alarm-deleted", slot = (slot as i64)));
227                                (*on_changed)();
228                            }
229                            Err(e) => status_label_del.set_label(&fl!("status-delete-failed", error = e.to_string())),
230                        }
231                        ControlFlow::Break
232                    }
233                    Err(TryRecvError::Empty) => ControlFlow::Continue,
234                    Err(TryRecvError::Disconnected) => {
235                        status_label_del.set_label(&fl!("status-delete-task-failed"));
236                        ControlFlow::Break
237                    }
238                });
239            });
240        }
241
242        let row_widgets: Vec<(TimeEntry, Switch, ToggleButton, ToggleButton, Vec<ToggleButton>)> = rows
243            .iter()
244            .map(|(_, w)| {
245                (
246                    w.time_entry.clone(),
247                    w.enabled_switch.clone(),
248                    w.snooze_toggle.clone(),
249                    w.once_toggle.clone(),
250                    w.day_toggles.clone(),
251                )
252            })
253            .collect();
254
255        let widget = Self {
256            container,
257            status_label: status_label.clone(),
258            refresh_button: refresh_button.clone(),
259            row_widgets,
260            manager: manager.clone(),
261            runtime: runtime.clone(),
262            connected_address: connected_address.clone(),
263        };
264
265        // Connect refresh button to read_alarms
266        {
267            let widget_manager = manager.clone();
268            let widget_runtime = runtime.clone();
269            let widget_addr = connected_address.clone();
270            let widget_status = status_label.clone();
271            let widget_rows = rows
272                .iter()
273                .map(|(_, w)| {
274                    (
275                        w.time_entry.clone(),
276                        w.enabled_switch.clone(),
277                        w.snooze_toggle.clone(),
278                        w.once_toggle.clone(),
279                        w.day_toggles.clone(),
280                    )
281                })
282                .collect::<Vec<_>>();
283            refresh_button.connect_clicked(move |_| {
284                let addr = *widget_addr.lock().unwrap_or_else(|p| {
285                    warn!("mutex poisoned - recovering");
286                    p.into_inner()
287                });
288                let Some(addr) = addr else {
289                    widget_status.set_label(&fl!("status-no-device-connected"));
290                    return;
291                };
292                widget_status.set_label(&fl!("status-reading-alarms"));
293                let manager = widget_manager.clone();
294                let (tx, rx) = std::sync::mpsc::channel::<Result<Vec<cgd1_rs::AlarmSlot>, String>>();
295                widget_runtime.spawn(async move {
296                    let result = async {
297                        let device = manager.device(&addr).await.ok_or_else(|| ClockError::Parse("device not found".into()))?;
298                        device.read_alarms().await
299                    }
300                    .await;
301                    let _ = tx.send(result.map_err(|e| e.to_string()));
302                });
303                let rx = std::cell::RefCell::new(rx);
304                let row_widgets = widget_rows.clone();
305                let status_label = widget_status.clone();
306                glib::source::idle_add_local(move || match rx.borrow_mut().try_recv() {
307                    Ok(result) => {
308                        match result {
309                            Ok(slots) => {
310                                for (i, (time_entry, enabled, snooze, once, days)) in row_widgets.iter().enumerate() {
311                                    let slot_alarm = slots.iter().find(|s| s.index.value() as usize == i);
312                                    match slot_alarm {
313                                        Some(s) => {
314                                            time_entry.set_time(s.entry.hour(), s.entry.minute());
315                                            enabled.set_active(s.entry.enabled());
316                                            snooze.set_active(s.entry.snooze());
317                                            let mask = s.entry.repeat_mask().value();
318                                            once.set_active(mask == DayMask::ONCE.value());
319                                            for (j, btn) in days.iter().enumerate() {
320                                                btn.set_active(mask & (1 << j) != 0);
321                                            }
322                                        }
323                                        None => {
324                                            enabled.set_active(false);
325                                        }
326                                    }
327                                }
328                                status_label.set_label(&fl!("status-loaded-alarms", count = (slots.len() as i64)));
329                            }
330                            Err(e) => {
331                                status_label.set_label(&fl!("status-read-failed", error = e.to_string()));
332                            }
333                        }
334                        ControlFlow::Break
335                    }
336                    Err(TryRecvError::Empty) => ControlFlow::Continue,
337                    Err(TryRecvError::Disconnected) => {
338                        status_label.set_label(&fl!("status-read-task-failed"));
339                        ControlFlow::Break
340                    }
341                });
342            });
343        }
344
345        widget
346    }

348    /// Read alarms from the connected device and update the UI.
349    pub fn read_alarms(&self) {
350        let addr = *self.connected_address.lock().unwrap_or_else(|p| {
351            warn!("mutex poisoned - recovering");
352            p.into_inner()
353        });
354        let Some(addr) = addr else {
355            self.status_label.set_label(&fl!("status-no-device-connected"));
356            return;
357        };
358        self.status_label.set_label(&fl!("status-reading-alarms"));
359        let manager = self.manager.clone();
360        let (tx, rx) = std::sync::mpsc::channel::<Result<Vec<AlarmSlot>, String>>();
361        self.runtime.spawn(async move {
362            let result = async {
363                let device = manager.device(&addr).await.ok_or_else(|| ClockError::Parse("device not found".into()))?;
364                device.read_alarms().await
365            }
366            .await;
367            let _ = tx.send(result.map_err(|e| e.to_string()));
368        });
369        let rx = std::cell::RefCell::new(rx);
370        let row_widgets = self.row_widgets.clone();
371        let status_label = self.status_label.clone();
372        glib::source::idle_add_local(move || match rx.borrow_mut().try_recv() {
373            Ok(result) => {
374                match result {
375                    Ok(slots) => {
376                        for (i, (time_entry, enabled, snooze, once, days)) in row_widgets.iter().enumerate() {
377                            let slot_alarm = slots.iter().find(|s| s.index.value() as usize == i);
378                            match slot_alarm {
379                                Some(s) => {
380                                    time_entry.set_time(s.entry.hour(), s.entry.minute());
381                                    enabled.set_active(s.entry.enabled());
382                                    snooze.set_active(s.entry.snooze());
383                                    let mask = s.entry.repeat_mask().value();
384                                    once.set_active(mask == DayMask::ONCE.value());
385                                    for (j, btn) in days.iter().enumerate() {
386                                        btn.set_active(mask & (1 << j) != 0);
387                                    }
388                                }
389                                None => {
390                                    enabled.set_active(false);
391                                }
392                            }
393                        }
394                        status_label.set_label(&fl!("status-loaded-alarms", count = (slots.len() as i64)));
395                    }
396                    Err(e) => {
397                        status_label.set_label(&fl!("status-read-failed", error = e.to_string()));
398                    }
399                }
400                ControlFlow::Break
401            }
402            Err(TryRecvError::Empty) => ControlFlow::Continue,
403            Err(TryRecvError::Disconnected) => {
404                status_label.set_label(&fl!("status-read-task-failed"));
405                ControlFlow::Break
406            }
407        });
408    }

411/// Create a single alarm slot row with time, repeat, enabled, and snooze controls.
412fn create_alarm_row() -> (Box, AlarmRowWidgets) {
413    let row = Box::builder().orientation(Orientation::Horizontal).spacing(4).build();
414
415    let enabled_switch = Switch::builder().tooltip_text(&fl!("tooltip-enable-alarm")).build();
416    enabled_switch.set_active(false);
417    row.append(&enabled_switch);
418
419    let time_entry = TimeEntry::new();
420    row.append(time_entry.widget());
421
422    let snooze_toggle = ToggleButton::builder()
423        .icon_name("nf-iec-sleep-mode-symbolic")
424        .tooltip_text(&fl!("tooltip-snooze"))
425        .build();
426    snooze_toggle.set_active(true);
427    row.append(&snooze_toggle);
428
429    let once_toggle = ToggleButton::builder()
430        .icon_name("nf-cod-calendar-symbolic")
431        .tooltip_text(&fl!("tooltip-once"))
432        .build();
433    once_toggle.set_active(true);
434    row.append(&once_toggle);
435
436    let day_labels = [
437        &fl!("day-mo"),
438        &fl!("day-tu"),
439        &fl!("day-we"),
440        &fl!("day-th"),
441        &fl!("day-fr"),
442        &fl!("day-sa"),
443        &fl!("day-su"),
444    ];
445    let day_toggles: Vec<ToggleButton> = day_labels.iter().map(|label| ToggleButton::builder().label(*label).build()).collect();
446
447    // Mutual exclusivity: Once vs day toggles
448    let once_for_days = once_toggle.clone();
449    let days_for_once = day_toggles.clone();
450    for btn in &day_toggles {
451        let once_clone = once_for_days.clone();
452        btn.connect_toggled(move |b| {
453            if b.is_active() && once_clone.is_active() {
454                once_clone.set_active(false);
455            }
456        });
457    }
458    once_toggle.connect_toggled(move |on| {
459        if on.is_active() {
460            for d in &days_for_once {
461                d.set_active(false);
462            }
463        }
464    });
465
466    for btn in &day_toggles {
467        row.append(btn);
468    }
469
470    let spacer = Box::builder().hexpand(true).build();
471    row.append(&spacer);
472
473    let set_button = Button::builder()
474        .icon_name("nf-cod-check-symbolic")
475        .tooltip_text(&fl!("tooltip-set"))
476        .css_classes(["suggested-action"])
477        .build();
478    row.append(&set_button);
479
480    let delete_button = Button::builder()
481        .icon_name("nf-cod-trash-symbolic")
482        .tooltip_text(&fl!("tooltip-delete"))
483        .css_classes(["destructive-action"])
484        .build();
485    row.append(&delete_button);
486
487    let widgets = AlarmRowWidgets {
488        time_entry,
489        enabled_switch,
490        snooze_toggle,
491        once_toggle,
492        day_toggles,
493        set_button,
494        delete_button,
495    };
496
497    (row, widgets)
