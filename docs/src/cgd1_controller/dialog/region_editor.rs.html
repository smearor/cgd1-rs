31/// Region editor widget for time format, temperature unit, language, and timezone.

33pub struct RegionEditorWidget {

38    /// Refresh button.
39    pub refresh_button: Button,
40    /// 24h format toggle.
41    time_format_24h: ToggleButton,
42    /// 12h format toggle.
43    time_format_12h: ToggleButton,
44    /// Celsius toggle.

47    temp_f: ToggleButton,
48    /// English language toggle.
49    lang_en: ToggleButton,
50    /// Chinese language toggle.
51    lang_zh: ToggleButton,
52    /// Timezone dropdown.
53    timezone_dropdown: DropDown,
54    /// Clock manager for device communication.
55    manager: Arc<ClockManager>,
56    /// Async runtime.
57    runtime: Arc<tokio::runtime::Runtime>,
58    /// Connected device address.
59    connected_address: Arc<std::sync::Mutex<Option<MacAddress>>>,
60}

98            .spacing(0)
99            .css_classes(["segmented"])

112        let temp_unit_box = Box::builder()
113            .orientation(Orientation::Horizontal)
114            .spacing(0)
115            .css_classes(["segmented"])
116            .build();
117        let temp_c = ToggleButton::builder().label("°C").css_classes(["segmented-btn"]).build();
118        let temp_f = ToggleButton::builder().label("°F").css_classes(["segmented-btn"]).build();
119        temp_c.set_active(true);
120        temp_c.set_group(Some(&temp_f));
121        temp_unit_box.append(&temp_c);
122        temp_unit_box.append(&temp_f);
123        temp_unit_row.append(&temp_unit_box);
124        temp_unit_row.append(&Box::builder().hexpand(true).build());
125        regional_box.append(&temp_unit_row);
126
127        let language_row = settings_row(&fl!("label-language"));
128        let language_box = Box::builder()
129            .orientation(Orientation::Horizontal)
130            .spacing(0)
131            .css_classes(["segmented"])
132            .build();
133        let lang_en = ToggleButton::builder().label(&fl!("toggle-english")).css_classes(["segmented-btn"]).build();
134        let lang_zh = ToggleButton::builder().label(&fl!("toggle-chinese")).css_classes(["segmented-btn"]).build();
135        lang_en.set_active(true);
136        lang_en.set_group(Some(&lang_zh));
137        language_box.append(&lang_en);
138        language_box.append(&lang_zh);
139        language_row.append(&language_box);
140        language_row.append(&Box::builder().hexpand(true).build());
141        regional_box.append(&language_row);
142
143        let timezone_entries = timezone_list();

160        regional_box.append(&sync_tz_button);

162        let tz_info_label = Label::builder()
163            .label(&fl!("info-tz-dst"))

217        // Read from Device
218        {
219            let manager = manager.clone();
220            let runtime = runtime.clone();
221            let connected_address = connected_address.clone();
222            let time_format_24h = time_format_24h.clone();
223            let time_format_12h = time_format_12h.clone();
224            let temp_c = temp_c.clone();
225            let temp_f = temp_f.clone();
226            let lang_en = lang_en.clone();
227            let lang_zh = lang_zh.clone();
228            let timezone_dropdown = timezone_dropdown.clone();
229            let status_label = status_label.clone();
230
231            refresh_button.connect_clicked(move |_| {

234                    p.into_inner()

236                let Some(addr) = addr else {
237                    status_label.set_label(&fl!("status-no-device-connected"));

247                    }

250                });
251                let rx = std::cell::RefCell::new(rx);
252                let time_format_24h = time_format_24h.clone();
253                let time_format_12h = time_format_12h.clone();
254                let temp_c = temp_c.clone();
255                let temp_f = temp_f.clone();
256                let lang_en = lang_en.clone();
257                let lang_zh = lang_zh.clone();
258                let timezone_dropdown = timezone_dropdown.clone();

268                                match settings.temperature_unit() {
269                                    TemperatureUnit::Celsius => temp_c.set_active(true),
270                                    TemperatureUnit::Fahrenheit => temp_f.set_active(true),
271                                }
272                                match settings.language() {
273                                    Language::English => lang_en.set_active(true),
274                                    Language::Chinese => lang_zh.set_active(true),
275                                }
276                                if let Some(idx) = find_timezone_index(settings.timezone().minutes()) {
277                                    timezone_dropdown.set_selected(idx);
278                                }
279                                status_label.set_label(&fl!("status-region-settings-loaded"));
280                            }
281                            Err(e) => {
282                                status_label.set_label(&fl!("status-read-failed", error = e.to_string()));
283                            }
284                        }
285                        glib::ControlFlow::Break
286                    }
287                    Err(std::sync::mpsc::TryRecvError::Empty) => glib::ControlFlow::Continue,
288                    Err(std::sync::mpsc::TryRecvError::Disconnected) => {
289                        status_label.set_label(&fl!("status-read-task-failed"));
290                        glib::ControlFlow::Break
291                    }
292                });
293            });
294        }
295
296        // Write to Device (read-modify-write: only region fields are modified)
297        {
298            let manager = manager.clone();
299            let runtime = runtime.clone();
300            let connected_address = connected_address.clone();
301            let time_format_12h = time_format_12h.clone();
302            let temp_f = temp_f.clone();
303            let lang_en = lang_en.clone();
304            let timezone_dropdown = timezone_dropdown.clone();
305            let status_label = status_label.clone();
306
307            apply_button.connect_clicked(move |_| {
308                let addr = *connected_address.lock().unwrap_or_else(|p| {
309                    warn!("mutex poisoned - recovering");
310                    p.into_inner()
311                });
312                let Some(addr) = addr else {
313                    status_label.set_label(&fl!("status-no-device-connected"));
314                    return;
315                };
316
317                let time_format = if time_format_12h.is_active() {
318                    TimeFormat::TwelveHour
319                } else {
320                    TimeFormat::TwentyFourHour
321                };
322                let temp_unit = if temp_f.is_active() {
323                    TemperatureUnit::Fahrenheit
324                } else {
325                    TemperatureUnit::Celsius
326                };
327                let language = if lang_en.is_active() { Language::English } else { Language::Chinese };
328                let tz_idx = timezone_dropdown.selected() as usize;
329                let timezone_entries = timezone_list();
330                let timezone_minutes = timezone_entries[tz_idx].1;
331                let timezone = match Timezone::from_minutes(timezone_minutes) {
332                    Ok(t) => t,
333                    Err(e) => {
334                        status_label.set_label(&fl!("status-invalid-timezone", error = e.to_string()));
335                        return;
336                    }
337                };
338
339                status_label.set_label(&fl!("status-writing-settings"));
340                let manager = manager.clone();
341                let (tx, rx) = std::sync::mpsc::channel::<Result<(), String>>();
342                runtime.spawn(async move {
343                    let result = async {
344                        let device = manager.device(&addr).await.ok_or_else(|| ClockError::Parse("device not found".into()))?;
345                        let current = device.read_settings().await?;
346                        let updated = DeviceSettings::new(
347                            current.volume(),
348                            time_format,
349                            temp_unit,
350                            language,
351                            timezone,
352                            current.screen_light_duration(),
353                            current.brightness(),
354                            current.night_brightness(),
355                            current.night_start(),
356                            current.night_end(),
357                            current.night_mode_enabled(),
358                            current.master_alarm_disabled(),
359                            current.ringtone_signature(),
360                        )?;
361                        device.write_settings(&updated).await
362                    }
363                    .await;
364                    let _ = tx.send(result.map_err(|e| e.to_string()));
365                });
366                let rx = std::cell::RefCell::new(rx);
367                let status_label = status_label.clone();
368                glib::source::idle_add_local(move || match rx.borrow_mut().try_recv() {
369                    Ok(result) => {
370                        match result {
371                            Ok(()) => status_label.set_label(&fl!("status-region-settings-written")),
372                            Err(e) => status_label.set_label(&fl!("status-write-failed", error = e.to_string())),
373                        }
374                        glib::ControlFlow::Break
375                    }
376                    Err(std::sync::mpsc::TryRecvError::Empty) => glib::ControlFlow::Continue,
377                    Err(std::sync::mpsc::TryRecvError::Disconnected) => {
378                        status_label.set_label(&fl!("status-write-task-failed"));
379                        glib::ControlFlow::Break
380                    }
381                });
382            });
383        }
384
385        Self {
386            container,
387            status_label: status_label.clone(),
388            refresh_button: refresh_button.clone(),
389            time_format_24h: time_format_24h.clone(),
390            time_format_12h: time_format_12h.clone(),
391            temp_c: temp_c.clone(),
392            temp_f: temp_f.clone(),
393            lang_en: lang_en.clone(),
394            lang_zh: lang_zh.clone(),
395            timezone_dropdown: timezone_dropdown.clone(),
396            manager: manager.clone(),
397            runtime: runtime.clone(),
398            connected_address: connected_address.clone(),
399        }

402    /// Read region settings from the connected device and update the UI.
403    pub fn read_settings(&self) {
404        let addr = *self.connected_address.lock().unwrap_or_else(|p| {
405            warn!("mutex poisoned - recovering");
406            p.into_inner()
407        });
408        let Some(addr) = addr else {
409            self.status_label.set_label(&fl!("status-no-device-connected"));
410            return;
411        };
412        self.status_label.set_label(&fl!("status-reading-settings"));
413        let manager = self.manager.clone();
414        let (tx, rx) = std::sync::mpsc::channel::<Result<DeviceSettings, String>>();
415        self.runtime.spawn(async move {
416            let result = async {
417                let device = manager.device(&addr).await.ok_or_else(|| ClockError::Parse("device not found".into()))?;
418                device.read_settings().await
419            }
420            .await;
421            let _ = tx.send(result.map_err(|e| e.to_string()));
422        });
423        let rx = std::cell::RefCell::new(rx);
424        let time_format_24h = self.time_format_24h.clone();
425        let time_format_12h = self.time_format_12h.clone();
426        let temp_c = self.temp_c.clone();
427        let temp_f = self.temp_f.clone();
428        let lang_en = self.lang_en.clone();
429        let lang_zh = self.lang_zh.clone();
430        let timezone_dropdown = self.timezone_dropdown.clone();
431        let status_label = self.status_label.clone();
432        glib::source::idle_add_local(move || match rx.borrow_mut().try_recv() {
433            Ok(result) => {
434                match result {
435                    Ok(settings) => {
436                        match settings.time_format() {
437                            TimeFormat::TwentyFourHour => time_format_24h.set_active(true),
438                            TimeFormat::TwelveHour => time_format_12h.set_active(true),
439                        }
440                        match settings.temperature_unit() {
441                            TemperatureUnit::Celsius => temp_c.set_active(true),
442                            TemperatureUnit::Fahrenheit => temp_f.set_active(true),
443                        }
444                        match settings.language() {
445                            Language::English => lang_en.set_active(true),
446                            Language::Chinese => lang_zh.set_active(true),
447                        }
448                        if let Some(idx) = find_timezone_index(settings.timezone().minutes()) {
449                            timezone_dropdown.set_selected(idx);
450                        }
451                        status_label.set_label(&fl!("status-region-settings-loaded"));
452                    }
453                    Err(e) => {
454                        status_label.set_label(&fl!("status-read-failed", error = e.to_string()));
455                    }
456                }
457                glib::ControlFlow::Break
458            }
459            Err(TryRecvError::Empty) => glib::ControlFlow::Continue,
460            Err(TryRecvError::Disconnected) => {
461                status_label.set_label(&fl!("status-read-task-failed"));
462                glib::ControlFlow::Break
463            }
464        });
465    }
