554    /// Set up the collapsible alarm editor panel with compact summary bar.
555    fn setup_alarm_panel(&self) {
556        let next_alarm_label = self.next_alarm_label.clone();
557        let device_states = self.device_states.clone();
558        let selected_address = self.selected_address.clone();
559        let manager_for_panel = self.manager.clone();
560        let runtime_for_panel = self.runtime.clone();
561        let editor = AlarmEditorWidget::new(self.manager.clone(), self.runtime.clone(), self.selected_address.clone(), move || {
562            let addr = *selected_address.lock().unwrap_or_else(|p| {
563                warn!("mutex poisoned - recovering");
564                p.into_inner()
565            });
566            let Some(addr) = addr else { return };
567            let (tx, rx) = std::sync::mpsc::channel::<Result<Vec<cgd1_rs::AlarmSlot>, String>>();
568            let manager = manager_for_panel.clone();
569            runtime_for_panel.spawn(async move {
570                let result = async {
571                    let device = manager.device(&addr).await.ok_or_else(|| ClockError::Parse("device not found".into()))?;
572                    device.read_alarms().await
573                }
574                .await;
575                let _ = tx.send(result.map_err(|e| e.to_string()));
576            });
577            let device_states = device_states.clone();
578            let next_alarm_label = next_alarm_label.clone();
579            let rx = std::cell::RefCell::new(rx);
580            glib::source::idle_add_local(move || match rx.borrow_mut().try_recv() {
581                Ok(result) => {
582                    match result {
583                        Ok(slots) => {
584                            let label_text = crate::next_alarm::next_alarm(&slots).map(|na| na.label.clone()).unwrap_or_default();
585                            next_alarm_label.set_label(&label_text);
586                            device_states
587                                .lock()
588                                .unwrap_or_else(|p| {
589                                    warn!("mutex poisoned - recovering");
590                                    p.into_inner()
591                                })
592                                .get_mut(&addr)
593                                .map(|s| s.alarms = Some(slots));
594                        }
595                        Err(e) => {
596                            warn!(%addr, error = %e, "on_changed: failed to re-read alarms");
597                        }
598                    }
599                    ControlFlow::Break
600                }
601                Err(TryRecvError::Empty) => ControlFlow::Continue,
602                Err(TryRecvError::Disconnected) => ControlFlow::Break,
603            });
604        });
605        self.alarm_revealer.set_child(Some(&editor.container));
606
607        let full_display = self.full_display.clone();
608        let summary_bar = self.summary_bar.clone();
609        let revealer = self.alarm_revealer.clone();
610        let display_toggle = self.display_toggle.clone();
611        let region_toggle = self.region_toggle.clone();
612        let sensor_toggle = self.sensor_toggle.clone();
613        let audio_toggle = self.audio_toggle.clone();
614        let editor = std::rc::Rc::new(editor);
615
616        self.alarm_toggle.connect_toggled(move |btn| {
617            let expanded = btn.is_active();
618            if expanded {
619                display_toggle.set_active(false);

1720    fn start_clock_tick(&self) {
1721        let date_display = self.date_display.clone();
1722        let time_display = self.time_display.clone();
1723        let temp_display = self.temp_display.clone();
1724        let humidity_display = self.humidity_display.clone();
1725        let summary_date = self.summary_date.clone();
1726        let summary_time = self.summary_time.clone();
1727
1728        let mut colon_visible = true;
1729
1730        glib::timeout_add_local(Duration::from_secs(1), move || {
1731            let now = match glib::DateTime::now_local()
1732                .or_else(|_| glib::DateTime::now_utc())
1733                .or_else(|_| glib::DateTime::from_unix_utc(0))
1734            {
1735                Ok(dt) => dt,
1736                Err(e) => {
1737                    error!(error = ?e, "all glib DateTime constructors failed, skipping clock tick");
1738                    return ControlFlow::Continue;
1739                }
1740            };
1741            let date_str = format!("{:04}-{:02}-{:02}", now.year(), now.month(), now.day_of_month());
1742            date_display.set_display_text(&date_str);
1743            summary_date.set_label(&date_str);
1744
1745            let sep = if colon_visible { ":" } else { " " };
1746            let time_str = format!("{:02}{sep}{:02}", now.hour(), now.minute());
1747            time_display.set_display_text(&time_str);
1748            summary_time.set_label(&time_str);
1749            colon_visible = !colon_visible;
1750
1751            if temp_display.imp().text.borrow().is_empty() {
1752                temp_display.set_display_text("--.-°C");
1753            }
1754            if humidity_display.imp().text.borrow().is_empty() {
1755                humidity_display.set_display_text("--.-%");
1756            }
1757
1758            ControlFlow::Continue
1759        });
1760    }

1763    fn populate_known_devices(&self) {

1778    /// Start periodic auto-scan with dynamic interval.
1779    ///
1780    /// - Disconnected: scan every 60 seconds
1781    /// - Connected: scan every 10 minutes
1782    /// Also checks for stale data (no sensor/battery events for 120 seconds)
1783    /// and disconnects the device if it appears unresponsive.
1784    fn start_auto_scan(&self) {
1785        let scan_btn = self.scan_button.clone();
1786        let selected_address = self.selected_address.clone();
1787        let device_states = self.device_states.clone();
1788        let last_scan_time = self.last_scan_time.clone();
1789        let connect_switch = self.connect_switch.clone();
1790        let status = self.status_label.clone();
1791        let dropdown = self.device_dropdown.clone();
1792        let known_devices = self.known_devices.clone();
1793        let known_device_store = self.known_device_store.clone();
1794
1795        // Try connecting to the first persisted known device immediately,
1796        // in parallel with the initial scan. The device may not be advertising
1797        // but BlueZ might still know it from a previous session.
1798        {
1799            let persisted = known_device_store.load();
1800            if let Some(first) = persisted.first() {
1801                debug!(address = %first, "controller: attempting startup connect to persisted known device");
1802                let known = known_devices.lock().unwrap_or_else(|p| {
1803                    warn!("mutex poisoned - recovering");
1804                    p.into_inner()
1805                });
1806                if let Some(idx) = known.iter().position(|a| a == first) {
1807                    dropdown.set_selected(idx as u32);
1808                    if !connect_switch.is_active() {
1809                        connect_switch.set_active(true);
1810                    }
1811                }
1812            }
1813        }
1814
1815        // Trigger initial scan immediately
1816        scan_btn.emit_clicked();
1817        *last_scan_time.lock().unwrap_or_else(|p| {
1818            warn!("mutex poisoned - recovering");
1819            p.into_inner()
1820        }) = Some(Instant::now());
1821
1822        // Periodic check: scan + staleness detection
1823        glib::timeout_add_local(Duration::from_secs(10), move || {
1824            let has_connected = device_states
1825                .lock()
1826                .unwrap_or_else(|p| {
1827                    warn!("mutex poisoned - recovering");
1828                    p.into_inner()
1829                })
1830                .values()
1831                .any(|s| s.connected);
1832            if has_connected {
1833                // Check all connected devices for staleness
1834                let stale_addrs: Vec<MacAddress> = {
1835                    let states = device_states.lock().unwrap_or_else(|p| {
1836                        warn!("mutex poisoned - recovering");
1837                        p.into_inner()
1838                    });
1839                    states
1840                        .iter()
1841                        .filter(|(_, state)| state.last_data_time.map_or(false, |t| t.elapsed() > Duration::from_secs(120)))
1842                        .map(|(addr, _)| *addr)
1843                        .collect()
1844                };
1845                if !stale_addrs.is_empty() {
1846                    for stale_addr in &stale_addrs {
1847                        warn!(address = %stale_addr, "controller: no data from device for 120s, disconnecting");
1848                    }
1849                    let first_stale = stale_addrs[0];
1850                    status.set_label(&fl!("status-device-unresponsive"));
1851                    if connect_switch.is_active() {
1852                        connect_switch.set_active(false);
1853                    }
1854                    // Mark stale devices as disconnected
1855                    {
1856                        let mut states = device_states.lock().unwrap_or_else(|p| {
1857                            warn!("mutex poisoned - recovering");
1858                            p.into_inner()
1859                        });
1860                        for addr in &stale_addrs {
1861                            if let Some(state) = states.get_mut(addr) {
1862                                state.connected = false;
1863                            }
1864                        }
1865                    }
1866                    *selected_address.lock().unwrap_or_else(|p| {
1867                        warn!("mutex poisoned - recovering");
1868                        p.into_inner()
1869                    }) = None;
1870
1871                    // Schedule a reconnect attempt after 5 seconds
1872                    let dropdown = dropdown.clone();
1873                    let connect_switch = connect_switch.clone();
1874                    let _selected_address = selected_address.clone();
1875                    let device_states = device_states.clone();
1876                    let known_devices = known_devices.clone();
1877                    let status = status.clone();
1878                    glib::timeout_add_local(Duration::from_secs(5), move || {
1879                        // Only reconnect if nothing is connected (e.g. by auto-scan)
1880                        if device_states
1881                            .lock()
1882                            .unwrap_or_else(|p| {
1883                                warn!("mutex poisoned - recovering");
1884                                p.into_inner()
1885                            })
1886                            .values()
1887                            .any(|s| s.connected)
1888                        {
1889                            debug!("controller: reconnect skipped, a device is connected");
1890                            return ControlFlow::Break;
1891                        }
1892                        // Find the device in the dropdown by address
1893                        let known = known_devices.lock().unwrap_or_else(|p| {
1894                            warn!("mutex poisoned - recovering");
1895                            p.into_inner()
1896                        });
1897                        if let Some(idx) = known.iter().position(|a| *a == first_stale) {
1898                            debug!(address = %first_stale, "controller: attempting reconnect to known device");
1899                            dropdown.set_selected(idx as u32);
1900                            status.set_label(&fl!("status-reconnecting"));
1901                            if !connect_switch.is_active() {
1902                                connect_switch.set_active(true);
1903                            }
1904                        } else {
1905                            warn!(address = %first_stale, "controller: reconnect failed, device no longer in known list");
1906                        }
1907                        ControlFlow::Break
1908                    });
1909                }
1910            }
1911
1912            let interval = if has_connected { Duration::from_secs(600) } else { Duration::from_secs(60) };
1913
1914            let should_scan = {
1915                let last = last_scan_time.lock().unwrap_or_else(|p| {
1916                    warn!("mutex poisoned - recovering");
1917                    p.into_inner()
1918                });
1919                match *last {
1920                    Some(t) => t.elapsed() >= interval,
1921                    None => true,
1922                }
1923            };
1924
1925            if should_scan {
1926                debug!(connected = has_connected, interval_secs = interval.as_secs(), "controller: triggering periodic scan");
1927                *last_scan_time.lock().unwrap_or_else(|p| {
1928                    warn!("mutex poisoned - recovering");
1929                    p.into_inner()
1930                }) = Some(Instant::now());
1931                scan_btn.emit_clicked();
1932            }
1933
1934            ControlFlow::Continue
1935        });
1936    }
1937}

1962/// Maps a battery level (0–100) to the appropriate Nerd Font battery icon name.
1963fn battery_icon_name(level: Option<u8>) -> &'static str {
1964    match level {
1965        Some(l) if l >= 87 => "nf-fa-battery-full-symbolic",
1966        Some(l) if l >= 62 => "nf-fa-battery-three-quarters-symbolic",
1967        Some(l) if l >= 37 => "nf-fa-battery-half-symbolic",
1968        Some(l) if l >= 12 => "nf-fa-battery-quarter-symbolic",
1969        Some(_) => "nf-fa-battery-empty-symbolic",
1970        None => "nf-fa-battery-empty-symbolic",
1971    }
1972}
