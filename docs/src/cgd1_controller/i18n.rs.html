1//! Internationalization support using the Fluent ecosystem.
2//!
3//! Embeds `.ftl` translation files at compile time via `rust-embed`
4//! and loads the system locale at runtime via `i18n-embed`.
5
6use std::sync::OnceLock;
7
8use i18n_embed::DesktopLanguageRequester;
9use i18n_embed::fluent::FluentLanguageLoader;
10use i18n_embed::fluent::fluent_language_loader;
11use rust_embed::RustEmbed;
12
13/// Embeds the `i18n/` directory containing `.ftl` files into the binary.
14#[derive(RustEmbed)]
15#[folder = "i18n/"]
16struct Localizations;
17
18/// The global fluent language loader.
19///
20/// Initialized once at first access with the system's preferred language,
21/// falling back to English (`en`) as the default/fallback language.
22static LOADER: OnceLock<FluentLanguageLoader> = OnceLock::new();
23
24/// Get the global language loader, initializing it on first access.
25pub fn loader() -> &'static FluentLanguageLoader {
26    LOADER.get_or_init(|| {
27        let loader = fluent_language_loader!();
28
29        let requested = DesktopLanguageRequester::requested_languages();
30        let languages = i18n_embed::select(&loader, &Localizations, &requested);
31
32        if let Err(error) = languages {
33            tracing::warn!(error = %error, "failed to load requested languages, falling back to default");
34        }
35
36        loader
37    })
38}

40/// Convenience macro wrapping `i18n_embed_fl::fl!` with the static loader.
41///
42/// Usage:
43/// ```
44/// use cgd1_rs_controller::fl;
45///
46/// let label = fl!("status-disconnected");
47/// ```
48#[macro_export]
49macro_rules! fl {
50    ($message_id:literal) => {{
51        i18n_embed_fl::fl!($crate::i18n::loader(), $message_id)
52    }};
53
54    ($message_id:literal, $($args:expr),*) => {{
55        i18n_embed_fl::fl!($crate::i18n::loader(), $message_id, $($args), *)
56    }};
57}
58
59/// Get a translated message by ID without compile-time checks.
60///
61/// Use this for dynamic message IDs (e.g. weekday names looked up by key).
62/// For static messages, prefer the `fl!` macro.
63pub fn get(message_id: &str) -> String {
64    loader().get(message_id)
65}
66
67/// Get a translated message with a single string argument.
68#[allow(dead_code)]
69pub fn get_str(message_id: &str, arg_name: &str, value: &str) -> String {
70    let mut args = fluent_bundle::FluentArgs::with_capacity(1);
71    args.set(arg_name, value);
72    loader().get_args_fluent(message_id, Some(&args))
73}
74
75/// Get a translated message with a single integer argument.
76pub fn get_int(message_id: &str, arg_name: &str, value: i64) -> String {
77    let mut args = fluent_bundle::FluentArgs::with_capacity(1);
78    args.set(arg_name, value);
79    loader().get_args_fluent(message_id, Some(&args))
80}

82/// Initialize the i18n system.
83///
84/// This forces the lazy loader to be initialized with the system locale.
85/// Call this early in `main()` to ensure translations are ready before
86/// any UI code runs.
87pub fn init() {
88    let _ = loader();
89}
