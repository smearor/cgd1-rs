8/// Events emitted by a connected CGD1 device.
9#[derive(Debug, Clone, PartialEq)]
10pub enum ClockEvent {
11    /// Real-time sensor update (temperature, humidity).
12    SensorUpdate {
13        /// Temperature in degrees Celsius.
14        temperature: Temperature,
15        /// Relative humidity in percent.
16        humidity: Humidity,
17    },
18    /// Battery level update.
19    BatteryLevel {
20        /// Battery percentage (0-100).
21        level: BatteryLevel,
22    },
23    /// ACK received for a command.
24    Ack {
25        /// Command ID that was acknowledged.
26        command: CommandId,
27        /// Status of the acknowledged command.
28        status: AckStatus,
29    },
30    /// Device disconnected.
31    Disconnected,
32    /// Device reconnected and state recovered (auth + subscriptions restored).
33    /// Callers that queued commands during the outage can retry.
34    Reconnected,
35    /// Passive advertisement received (no connection needed).
36    Advertisement(AdvertisementData),
37}
