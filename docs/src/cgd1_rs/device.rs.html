39/// Shared state for matching ACKs to pending requests.
40///
41/// Uses a `VecDeque` per command byte so that multiple concurrent requests
42/// with the same command byte are queued rather than overwriting each other.
43/// The notification task pops the front sender when an ACK arrives.
44type PendingMap = Arc<Mutex<HashMap<CommandId, VecDeque<oneshot::Sender<Result<Ack>>>>>>;
45
46/// Register a pending request for a given command byte.
47///
48/// Pushes the oneshot sender onto the queue for that command. When the
49/// notification task receives an ACK with a matching command byte, it pops
50/// the front sender and delivers the result.
51async fn register_pending(pending: &PendingMap, command: CommandId, sender: oneshot::Sender<Result<Ack>>) {
52    let mut map = pending.lock().await;
53    map.entry(command).or_default().push_back(sender);
54}
55
56/// Pop the next live pending sender for a given command byte.
57///
58/// Called by the notification task when an ACK arrives. Skips senders whose
59/// receivers have been dropped (e.g., due to timeout). Returns `None` if
60/// no live request is pending for that command.
61async fn pop_pending_alive(pending: &PendingMap, command: CommandId) -> Option<oneshot::Sender<Result<Ack>>> {
62    let mut map = pending.lock().await;
63    let queue = map.get_mut(&command)?;
64    while let Some(sender) = queue.pop_front() {
65        if !sender.is_closed() {
66            return Some(sender);
67        }
68    }
69    None
70}

72/// Handle to a connected Qingping CGD1 alarm clock.
73///
74/// All command methods that send a frame and wait for an ACK are serialized
75/// via `command_mutex` to prevent race conditions when multiple callers
76/// invoke the same command byte concurrently.
77///
78/// Stores the last successful auth token so that reconnection can
79/// automatically re-authenticate without caller intervention.
80#[derive(Clone)]
81pub struct ClockDevice {
82    transport: Arc<dyn BleTransport>,
83    address: MacAddress,
84    event_sender: broadcast::Sender<ClockEvent>,
85    /// Serializes write operations to prevent concurrent same-command races.
86    command_mutex: Arc<Mutex<()>>,
87    /// Last successful auth token, used for automatic re-auth on reconnect.
88    auth_token: Arc<Mutex<Option<AuthToken>>>,
89    /// Whether the device is currently authenticated.
90    is_authenticated: Arc<AtomicBool>,
91    /// Pending request-response channels keyed by command byte.
92    pending: PendingMap,
93    /// Pending channel for non-ACK data notifications (e.g. firmware, alarms).
94    /// Uses mpsc to support multi-packet responses.
95    pending_data_response: Arc<Mutex<Option<mpsc::Sender<Vec<u8>>>>>,
96    /// Optional token store for persisting auth tokens after successful privileged commands.
97    token_store: Arc<Mutex<Option<Arc<dyn TokenStore>>>>,
98}

145    ///
146    /// The task listens for BLE notifications, dispatches ACKs to pending
147    /// request-response channels, broadcasts sensor/battery events, and

150        let transport = self.transport.clone();
151        let event_sender = self.event_sender.clone();
152        let pending = self.pending.clone();
153        let is_authenticated = self.is_authenticated.clone();
154        let auth_token = self.auth_token.clone();

161            notification_task(
162                transport,
163                event_sender,
164                pending,
165                is_authenticated,
166                auth_token,

168                command_mutex,

170                token_store,
171            )
172            .await
173        });
174    }

176    /// Wait for an ACK with the given command byte.
177    ///
178    /// Registers a pending request and waits up to `RESPONSE_TIMEOUT_SECS`
179    /// for the notification task to deliver the ACK.
180    async fn wait_for_ack(&self, command: Command) -> Result<Ack> {
181        let command_id = command.command_id();
182        let (sender, receiver) = oneshot::channel();
183        register_pending(&self.pending, command_id, sender).await;
184
185        match timeout(Duration::from_secs(RESPONSE_TIMEOUT_SECS), receiver).await {
186            Ok(Ok(ack)) => ack,
187            Ok(Err(_)) => Err(ClockError::Transport("pending request canceled".into())),
188            Err(_) => Err(ClockError::Timeout),
189        }
190    }

192    /// Authenticate with the device using a 16-byte token.
193    ///
194    /// Performs the two-step handshake:
195    /// 1. Send Auth Init: `11 01 [Token 16B]` to Auth Write.
196    /// 2. Wait for ACK on Auth Notify: `04 ff 01 00 [Payload]`.
197    /// 3. Send Auth Confirm: `11 02 [Token 16B]` to Auth Write.
198    /// 4. Wait for final ACK: `04 ff 02 00 00`.
199    pub async fn authenticate(&self, token: &AuthToken) -> Result<()> {
200        let _guard = self.command_mutex.lock().await;
201
202        // Step 1: Auth Init
203        self.transport.write_frame(Command::AuthInit, token.payload()).await?;
204
205        let ack = self.wait_for_ack(Command::AuthInit).await?;
206        if let AckStatus::Failure(code) = ack.status {
207            return Err(ClockError::AuthFailed(format!("init status: {code:#04x}")));
208        }
209
210        // Step 2: Auth Confirm
211        self.transport.write_frame(Command::AuthConfirm, token.payload()).await?;
212
213        let ack = self.wait_for_ack(Command::AuthConfirm).await?;
214        if let AckStatus::Failure(code) = ack.status {
215            return Err(ClockError::AuthFailed(format!("confirm status: {code:#04x}")));
216        }
217
218        // Store the token for automatic re-auth on reconnect.
219        {
220            let mut token_guard = self.auth_token.lock().await;
221            *token_guard = Some(token.clone());
222        }
223        self.is_authenticated.store(true, Ordering::SeqCst);
224
225        Ok(())
226    }

228    /// Synchronize the device clock to the given Unix timestamp.
229    ///
230    /// Sends: `05 09 [Timestamp 4B LE]` to Auth Write.
231    /// Expects ACK: `04 ff 09 00 00`.
232    ///
233    /// This is the first privileged command after authentication. If the
234    /// token was rejected, the device will drop the connection here.
235    pub async fn sync_time(&self, timestamp: u32) -> Result<()> {
236        let _guard = self.command_mutex.lock().await;
237
238        let payload = timestamp.to_le_bytes();
239        self.transport.write_frame(Command::TimeSync, &payload).await?;
240
241        let ack = self.wait_for_ack(Command::TimeSync).await?;
242        if let AckStatus::Failure(_) = ack.status {
243            return Err(ClockError::CommandRejected {
244                command: 0x09,
245                status: ack.status,
246            });
247        }
248
249        // Token is now confirmed — persist it if a token store is configured.
250        let store = self.token_store.lock().await;
251        if let Some(ref store) = *store {
252            let token = self.auth_token.lock().await;
253            if let Some(ref token) = *token {
254                store.save(&self.address, token)?;
255            }
256        }
257
258        Ok(())
259    }

267    /// Read the firmware version string from the device.
268    ///
269    /// Sends: `01 0d` to Auth Write.
270    /// Expects response on Auth Notify: `0b [Byte] [ASCII String]`.
271    pub async fn read_firmware(&self) -> Result<String> {
272        let _guard = self.command_mutex.lock().await;
273
274        // Set up a pending mpsc channel for non-ACK data notifications.
275        let (sender, mut receiver) = mpsc::channel(16);
276        {
277            let mut pending = self.pending_data_response.lock().await;
278            *pending = Some(sender);
279        }
280
281        self.transport.write_frame(Command::ReadFirmware, &[]).await?;
282
283        let response = match timeout(Duration::from_secs(RESPONSE_TIMEOUT_SECS), receiver.recv()).await {
284            Ok(Some(data)) => data,
285            Ok(None) => return Err(ClockError::Transport("firmware response canceled".into())),
286            Err(_) => {
287                let mut pending = self.pending_data_response.lock().await;
288                *pending = None;
289                return Err(ClockError::Timeout);
290            }
291        };
292
293        // Clean up the pending channel.
294        {
295            let mut pending = self.pending_data_response.lock().await;
296            *pending = None;
297        }
298
299        // Parse: skip length byte, skip one byte, rest is ASCII string.
300        if response.len() < 2 {
301            return Err(ClockError::Parse("firmware response too short".into()));
302        }
303        let version = String::from_utf8_lossy(&response[2..]).to_string();
304        Ok(version)
305    }

349    /// Read all alarm slots from the device.
350    ///
351    /// Sends: `01 06` to Data Write.
352    /// Expects multiple response packets on Data Notify, each containing
353    /// up to 3 alarm entries (5 bytes each). The device sends 6 packets
354    /// for all 16 slots.
355    pub async fn read_alarms(&self) -> Result<Vec<AlarmSlot>> {
356        let _guard = self.command_mutex.lock().await;
357
358        // Set up a pending mpsc channel for multi-packet data notifications.
359        let (sender, mut receiver) = mpsc::channel(16);
360        {
361            let mut pending = self.pending_data_response.lock().await;
362            *pending = Some(sender);
363        }
364
365        self.transport.write_frame(Command::ReadAlarms, &[]).await?;
366
367        // Collect all data packets within the timeout period.
368        // The device sends 6 packets for 16 slots (3 per packet).
369        let mut all_data = Vec::new();
370        let deadline = Duration::from_secs(RESPONSE_TIMEOUT_SECS);
371
372        loop {
373            match timeout(deadline, receiver.recv()).await {
374                Ok(Some(data)) => {
375                    all_data.push(data);
376                    // If we've received 6 packets, we have all slots.
377                    if all_data.len() >= 6 {
378                        break;
379                    }
380                }
381                Ok(None) => break,
382                Err(_) => break,
383            }
384        }
385
386        // Clean up the pending channel.
387        {
388            let mut pending = self.pending_data_response.lock().await;
389            *pending = None;
390        }
391
392        if all_data.is_empty() {
393            return Err(ClockError::Timeout);
394        }
395
396        // Parse all received packets.
397        // Each packet: [Length] [0x06] [BaseIndex] [Entry1 5B] [Entry2 5B] [Entry3 5B]
398        let mut slots = Vec::new();
399        for packet in &all_data {
400            if packet.len() < 3 {
401                continue;
402            }
403            // Skip length byte (packet[0]) and command byte (packet[1]).
404            // packet[2] is the base index (slot index of the first entry in this packet).
405            let base_index = packet[2] as usize;
406            let entries_data = &packet[3..];
407
408            for (i, chunk) in entries_data.chunks(5).enumerate() {
409                if chunk.len() < 5 {
410                    break;
411                }
412                let raw_slot = (base_index + i) as u8;
413                let Ok(index) = AlarmSlotIndex::new(raw_slot) else {
414                    continue;
415                };
416                if let Some(entry) = AlarmEntry::decode(chunk)? {
417                    slots.push(AlarmSlot { index, entry });
418                }
419            }
420        }
421
422        Ok(slots)
423    }

434/// Parse sensor data from a Sensor Notify notification.
435///
436/// Format: `[00] [Temp L] [Temp H] [Hum L] [Hum H]` (5 bytes).
437fn parse_sensor_data(value: &[u8]) -> Option<(Temperature, Humidity)> {
438    if value.len() >= 5 {
439        let temp = i16::from_le_bytes([value[1], value[2]]) as f32 / 100.0;
440        let humidity = u16::from_le_bytes([value[3], value[4]]) as f32 / 100.0;
441        Some((Temperature::new(temp), Humidity::new(humidity)))
442    } else {
443        None
444    }
445}

447/// Background notification task.
448///
449/// Listens for BLE notifications from the connected device, dispatches ACKs
450/// to pending request-response channels, broadcasts sensor/battery events,
451/// and attempts reconnection on disconnect.
452#[allow(clippy::too_many_arguments)]
453async fn notification_task(
454    transport: Arc<dyn BleTransport>,
455    event_sender: broadcast::Sender<ClockEvent>,
456    pending: PendingMap,
457    is_authenticated: Arc<AtomicBool>,
458    auth_token: Arc<Mutex<Option<AuthToken>>>,
459    address: MacAddress,
460    command_mutex: Arc<Mutex<()>>,
461    pending_data_response: Arc<Mutex<Option<mpsc::Sender<Vec<u8>>>>>,
462    token_store: Arc<Mutex<Option<Arc<dyn TokenStore>>>>,
463) {
464    let sensor_uuid = CharacteristicUuid::SensorNotify.uuid();
465    let battery_uuid = CharacteristicUuid::BatteryLevel.uuid();
466
467    loop {
468        match transport.next_notification().await {
469            Some((uuid, value)) => {
470                if uuid == sensor_uuid {
471                    if let Some((temp, humidity)) = parse_sensor_data(&value) {
472                        let _ = event_sender.send(ClockEvent::SensorUpdate { temperature: temp, humidity });
473                    }
474                } else if uuid == battery_uuid {
475                    if let Some(&level) = value.first() {
476                        let _ = event_sender.send(ClockEvent::BatteryLevel {
477                            level: BatteryLevel::new(level),
478                        });
479                    }
480                } else if let Some(ack) = Ack::parse(&value) {
481                    let _ = event_sender.send(ClockEvent::Ack {
482                        command: ack.command,
483                        status: ack.status,
484                    });
485
486                    if let Some(sender) = pop_pending_alive(&pending, ack.command).await {
487                        let _ = sender.send(Ok(ack));
488                    }
489                } else {
490                    // Non-ACK data notification — forward to pending data response channel.
491                    let sender = {
492                        let pending = pending_data_response.lock().await;
493                        pending.clone()
494                    };
495                    if let Some(sender) = sender {
496                        let _ = sender.send(value).await;
497                    } else {
498                        debug!(
499                            uuid = %uuid,
500                            len = value.len(),
501                            "unhandled notification, ignoring"
502                        );
503                    }
504                }
505            }
506            None => {
507                warn!("notification stream ended, device disconnected");
508                let _ = event_sender.send(ClockEvent::Disconnected);
509                is_authenticated.store(false, Ordering::SeqCst);
510
511                let device = ClockDevice {
512                    transport: transport.clone(),
513                    address,
514                    event_sender: event_sender.clone(),
515                    command_mutex: command_mutex.clone(),
516                    auth_token: auth_token.clone(),
517                    is_authenticated: is_authenticated.clone(),
518                    pending: pending.clone(),
519                    pending_data_response: pending_data_response.clone(),
520                    token_store: token_store.clone(),
521                };
522
523                match reconnect_and_restore(&device, 6).await {
524                    Ok(()) => {
525                        info!("reconnect successful, resuming notification task");
526                        continue;
527                    }
528                    Err(e) => {
529                        warn!("reconnect failed after all attempts: {e:?}");
530                        break;
531                    }
532                }
533            }
534        }
535    }
536}

538/// Reconnect with exponential backoff and full state recovery.
539///
540/// Delay sequence: 1s, 2s, 4s, 8s, 16s, 32s (capped).
541///
542/// After a successful BLE connect, re-subscribes to all GATT notify
543/// characteristics and re-authenticates with the stored token. This
544/// ensures the device is fully operational before commands resume.
545async fn reconnect_and_restore(device: &ClockDevice, max_attempts: u32) -> Result<()> {
546    let mut delay = Duration::from_secs(1);
547    let max_delay = Duration::from_secs(32);
548
549    for attempt in 1..=max_attempts {
550        debug!(attempt, delay_ms = delay.as_millis(), "attempting reconnect");
551
552        // Step 1: BLE Reconnect
553        if device.transport.connect(&device.address).await.is_err() {
554            tokio::time::sleep(delay).await;
555            delay = (delay * 2).min(max_delay);
556            continue;
557        }
558
559        // Step 2: Re-subscribe to all notify characteristics
560        let characteristics = [CharacteristicUuid::AuthNotify, CharacteristicUuid::DataNotify, CharacteristicUuid::SensorNotify];
561
562        let mut all_subscribed = true;
563        for char_uuid in &characteristics {
564            if device.transport.subscribe(*char_uuid).await.is_err() {
565                all_subscribed = false;
566            }
567        }
568
569        if !all_subscribed {
570            warn!("reconnect: GATT re-subscription failed, retrying");
571            tokio::time::sleep(delay).await;
572            delay = (delay * 2).min(max_delay);
573            continue;
574        }
575
576        // Step 3: Re-authenticate with stored token
577        let token = device.auth_token.lock().await;
578        if let Some(ref token) = *token
579            && device.authenticate(token).await.is_err()
580        {
581            warn!("reconnect: re-authentication failed, retrying");
582            tokio::time::sleep(delay).await;
583            delay = (delay * 2).min(max_delay);
584            continue;
585        }
586
587        // Step 4: Mark as authenticated and notify subscribers
588        device.is_authenticated.store(true, Ordering::SeqCst);
589        let _ = device.event_sender.send(ClockEvent::Reconnected);
590        info!("reconnect: state recovery complete");
591        return Ok(());
592    }
593
594    Err(ClockError::Transport("reconnect with state recovery failed".into()))
595}

681    #[tokio::test]
682    async fn read_alarms_parses_multi_packet() {
683        let mock = Arc::new(MockBleTransport::new());
684        let addr = MacAddress::parse("AA:BB:CC:DD:EE:FF").unwrap();
685        let device = ClockDevice::new(mock.clone(), addr);
686        device.spawn_notification_task();
687
688        // Pre-push 6 data notify packets on Data Notify.
689        // The auto-ACK for the ReadAlarms write will arrive first, then these data packets.
690        // Each packet: [length] [0x06] [base_index] [entry1 5B] [entry2 5B] [entry3 5B]
691        let data_notify = CharacteristicUuid::DataNotify.uuid();
692
693        // Packet 0: slots 0-2, slot 0 has alarm at 07:30 weekdays
694        mock.push_notification(
695            data_notify,
696            vec![
697                0x11, 0x06, 0x00, 0x01, 0x07, 0x1E, 0x3E, 0x01, // slot 0: enabled, 7:30, weekdays, snooze
698                0xFF, 0xFF, 0xFF, 0xFF, 0xFF, // slot 1: empty
699                0xFF, 0xFF, 0xFF, 0xFF, 0xFF, // slot 2: empty
700            ],
701        );
702        // Packets 1-5: all empty slots
703        for base in [3u8, 6, 9, 12, 15] {
704            mock.push_notification(
705                data_notify,
706                vec![
707                    0x11, 0x06, base, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF,
708                ],
709            );
710        }
711
712        let slots = device.read_alarms().await.unwrap();
713        assert_eq!(slots.len(), 1);
714        assert_eq!(slots[0].index.value(), 0);
715        assert_eq!(slots[0].entry.hour(), 7);
716        assert_eq!(slots[0].entry.minute(), 30);
717        assert_eq!(slots[0].entry.repeat_mask(), DayMask::WEEKDAYS);
718        assert!(slots[0].entry.enabled());
719        assert!(slots[0].entry.snooze());
720    }
