32/// Response timeout in seconds.
33const RESPONSE_TIMEOUT_SECS: u64 = 10;

35/// Shared state for matching ACKs to pending requests.
36///
37/// Uses a `VecDeque` per command byte so that multiple concurrent requests
38/// with the same command byte are queued rather than overwriting each other.
39/// The notification task pops the front sender when an ACK arrives.
40type PendingMap = Arc<Mutex<HashMap<CommandId, VecDeque<oneshot::Sender<Result<Ack>>>>>>;
41
42/// Register a pending request for a given command byte.
43///
44/// Pushes the oneshot sender onto the queue for that command. When the
45/// notification task receives an ACK with a matching command byte, it pops
46/// the front sender and delivers the result.
47async fn register_pending(pending: &PendingMap, command: CommandId, sender: oneshot::Sender<Result<Ack>>) {
48    let mut map = pending.lock().await;
49    map.entry(command).or_default().push_back(sender);
50}
51
52/// Pop the next live pending sender for a given command byte.
53///
54/// Called by the notification task when an ACK arrives. Skips senders whose
55/// receivers have been dropped (e.g., due to timeout). Returns `None` if
56/// no live request is pending for that command.
57async fn pop_pending_alive(pending: &PendingMap, command: CommandId) -> Option<oneshot::Sender<Result<Ack>>> {
58    let mut map = pending.lock().await;
59    let queue = map.get_mut(&command)?;
60    while let Some(sender) = queue.pop_front() {
61        if !sender.is_closed() {
62            return Some(sender);
63        }
64    }
65    None
66}

68/// Handle to a connected Qingping CGD1 alarm clock.
69///
70/// All command methods that send a frame and wait for an ACK are serialized
71/// via `command_mutex` to prevent race conditions when multiple callers
72/// invoke the same command byte concurrently.
73///
74/// Stores the last successful auth token so that reconnection can
75/// automatically re-authenticate without caller intervention.
76#[derive(Clone)]
77pub struct ClockDevice {
78    transport: Arc<dyn BleTransport>,
79    address: MacAddress,
80    event_sender: broadcast::Sender<ClockEvent>,
81    /// Serializes write operations to prevent concurrent same-command races.
82    command_mutex: Arc<Mutex<()>>,
83    /// Last successful auth token, used for automatic re-auth on reconnect.
84    auth_token: Arc<Mutex<Option<AuthToken>>>,
85    /// Whether the device is currently authenticated.
86    is_authenticated: Arc<AtomicBool>,
87    /// Pending request-response channels keyed by command byte.
88    pending: PendingMap,
89    /// Pending channel for the next non-ACK data notification (e.g. firmware).
90    pending_data_response: Arc<Mutex<Option<oneshot::Sender<Vec<u8>>>>>,
91    /// Optional token store for persisting auth tokens after successful privileged commands.
92    token_store: Arc<Mutex<Option<Arc<dyn TokenStore>>>>,
93}

139    /// Spawn the background notification task for this device.
140    ///
141    /// The task listens for BLE notifications, dispatches ACKs to pending
142    /// request-response channels, broadcasts sensor/battery events, and
143    /// attempts reconnection on disconnect.
144    pub fn spawn_notification_task(&self) {
145        let transport = self.transport.clone();
146        let event_sender = self.event_sender.clone();
147        let pending = self.pending.clone();
148        let is_authenticated = self.is_authenticated.clone();
149        let auth_token = self.auth_token.clone();
150        let address = self.address;
151        let command_mutex = self.command_mutex.clone();
152        let pending_data_response = self.pending_data_response.clone();
153        let token_store = self.token_store.clone();
154
155        tokio::spawn(async move {
156            notification_task(
157                transport,
158                event_sender,
159                pending,
160                is_authenticated,
161                auth_token,
162                address,
163                command_mutex,
164                pending_data_response,
165                token_store,
166            )
167            .await
168        });
169    }

171    /// Wait for an ACK with the given command byte.
172    ///
173    /// Registers a pending request and waits up to `RESPONSE_TIMEOUT_SECS`
174    /// for the notification task to deliver the ACK.
175    async fn wait_for_ack(&self, command: Command) -> Result<Ack> {
176        let command_id = command.command_id();
177        let (sender, receiver) = oneshot::channel();
178        register_pending(&self.pending, command_id, sender).await;
179
180        match timeout(Duration::from_secs(RESPONSE_TIMEOUT_SECS), receiver).await {
181            Ok(Ok(ack)) => ack,
182            Ok(Err(_)) => Err(ClockError::Transport("pending request canceled".into())),
183            Err(_) => Err(ClockError::Timeout),
184        }
185    }
186
187    /// Authenticate with the device using a 16-byte token.
188    ///
189    /// Performs the two-step handshake:
190    /// 1. Send Auth Init: `11 01 [Token 16B]` to Auth Write.
191    /// 2. Wait for ACK on Auth Notify: `04 ff 01 00 [Payload]`.
192    /// 3. Send Auth Confirm: `11 02 [Token 16B]` to Auth Write.
193    /// 4. Wait for final ACK: `04 ff 02 00 00`.
194    pub async fn authenticate(&self, token: &AuthToken) -> Result<()> {
195        let _guard = self.command_mutex.lock().await;
196
197        // Step 1: Auth Init
198        self.transport.write_frame(Command::AuthInit, token.payload()).await?;
199
200        let ack = self.wait_for_ack(Command::AuthInit).await?;
201        if let AckStatus::Failure(code) = ack.status {
202            return Err(ClockError::AuthFailed(format!("init status: {code:#04x}")));
203        }
204
205        // Step 2: Auth Confirm
206        self.transport.write_frame(Command::AuthConfirm, token.payload()).await?;
207
208        let ack = self.wait_for_ack(Command::AuthConfirm).await?;
209        if let AckStatus::Failure(code) = ack.status {
210            return Err(ClockError::AuthFailed(format!("confirm status: {code:#04x}")));
211        }
212
213        // Store the token for automatic re-auth on reconnect.
214        {
215            let mut token_guard = self.auth_token.lock().await;
216            *token_guard = Some(token.clone());
217        }
218        self.is_authenticated.store(true, Ordering::SeqCst);
219
220        Ok(())
221    }

223    /// Synchronize the device clock to the given Unix timestamp.
224    ///
225    /// Sends: `05 09 [Timestamp 4B LE]` to Auth Write.
226    /// Expects ACK: `04 ff 09 00 00`.
227    ///
228    /// This is the first privileged command after authentication. If the
229    /// token was rejected, the device will drop the connection here.
230    pub async fn sync_time(&self, timestamp: u32) -> Result<()> {
231        let _guard = self.command_mutex.lock().await;
232
233        let payload = timestamp.to_le_bytes();
234        self.transport.write_frame(Command::TimeSync, &payload).await?;
235
236        let ack = self.wait_for_ack(Command::TimeSync).await?;
237        if let AckStatus::Failure(_) = ack.status {
238            return Err(ClockError::CommandRejected {
239                command: 0x09,
240                status: ack.status,
241            });
242        }
243
244        // Token is now confirmed — persist it if a token store is configured.
245        let store = self.token_store.lock().await;
246        if let Some(ref store) = *store {
247            let token = self.auth_token.lock().await;
248            if let Some(ref token) = *token {
249                store.save(&self.address, token)?;
250            }
251        }
252
253        Ok(())
254    }
255
256    /// Synchronize the device clock to the current system time.
257    pub async fn sync_time_now(&self) -> Result<()> {
258        let now = SystemTime::now()
259            .duration_since(UNIX_EPOCH)
260            .map_err(|e| ClockError::Transport(e.to_string()))?;
261        self.sync_time(now.as_secs() as u32).await
262    }

264    /// Read the firmware version string from the device.
265    ///
266    /// Sends: `01 0d` to Auth Write.
267    /// Expects response on Auth Notify: `0b [Byte] [ASCII String]`.
268    pub async fn read_firmware(&self) -> Result<String> {
269        let _guard = self.command_mutex.lock().await;
270
271        // Set up a pending channel for the non-ACK data notification.
272        let (sender, receiver) = oneshot::channel();
273        {
274            let mut pending = self.pending_data_response.lock().await;
275            *pending = Some(sender);
276        }
277
278        self.transport.write_frame(Command::ReadFirmware, &[]).await?;
279
280        let response = match timeout(Duration::from_secs(RESPONSE_TIMEOUT_SECS), receiver).await {
281            Ok(Ok(data)) => data,
282            Ok(Err(_)) => return Err(ClockError::Transport("firmware response canceled".into())),
283            Err(_) => {
284                let mut pending = self.pending_data_response.lock().await;
285                *pending = None;
286                return Err(ClockError::Timeout);
287            }
288        };
289
290        // Parse: skip length byte, skip one byte, rest is ASCII string.
291        if response.len() < 2 {
292            return Err(ClockError::Parse("firmware response too short".into()));
293        }
294        let version = String::from_utf8_lossy(&response[2..]).to_string();
295        Ok(version)
296    }

298    /// Disconnect from the device.
299    pub async fn disconnect(&self) -> Result<()> {
300        self.transport.disconnect().await?;
301        self.is_authenticated.store(false, Ordering::SeqCst);
302        let _ = self.event_sender.send(ClockEvent::Disconnected);
303        Ok(())
304    }

307/// Parse sensor data from a Sensor Notify notification.
308///
309/// Format: `[00] [Temp L] [Temp H] [Hum L] [Hum H]` (5 bytes).
310fn parse_sensor_data(value: &[u8]) -> Option<(Temperature, Humidity)> {
311    if value.len() >= 5 {
312        let temp = i16::from_le_bytes([value[1], value[2]]) as f32 / 100.0;
313        let humidity = u16::from_le_bytes([value[3], value[4]]) as f32 / 100.0;
314        Some((Temperature::new(temp), Humidity::new(humidity)))
315    } else {
316        None
317    }
318}

320/// Background notification task.
321///
322/// Listens for BLE notifications from the connected device, dispatches ACKs
323/// to pending request-response channels, broadcasts sensor/battery events,
324/// and attempts reconnection on disconnect.
325#[allow(clippy::too_many_arguments)]
326async fn notification_task(
327    transport: Arc<dyn BleTransport>,
328    event_sender: broadcast::Sender<ClockEvent>,
329    pending: PendingMap,
330    is_authenticated: Arc<AtomicBool>,
331    auth_token: Arc<Mutex<Option<AuthToken>>>,
332    address: MacAddress,
333    command_mutex: Arc<Mutex<()>>,
334    pending_data_response: Arc<Mutex<Option<oneshot::Sender<Vec<u8>>>>>,
335    token_store: Arc<Mutex<Option<Arc<dyn TokenStore>>>>,
336) {
337    let sensor_uuid = CharacteristicUuid::SensorNotify.uuid();
338    let battery_uuid = CharacteristicUuid::BatteryLevel.uuid();
339
340    loop {
341        match transport.next_notification().await {
342            Some((uuid, value)) => {
343                if uuid == sensor_uuid {
344                    if let Some((temp, humidity)) = parse_sensor_data(&value) {
345                        let _ = event_sender.send(ClockEvent::SensorUpdate { temperature: temp, humidity });
346                    }
347                } else if uuid == battery_uuid {
348                    if let Some(&level) = value.first() {
349                        let _ = event_sender.send(ClockEvent::BatteryLevel {
350                            level: BatteryLevel::new(level),
351                        });
352                    }
353                } else if let Some(ack) = Ack::parse(&value) {
354                    let _ = event_sender.send(ClockEvent::Ack {
355                        command: ack.command,
356                        status: ack.status,
357                    });
358
359                    if let Some(sender) = pop_pending_alive(&pending, ack.command).await {
360                        let _ = sender.send(Ok(ack));
361                    }
362                } else {
363                    // Non-ACK data notification — check for a pending data response request.
364                    let sender = {
365                        let mut pending = pending_data_response.lock().await;
366                        pending.take()
367                    };
368                    if let Some(sender) = sender {
369                        let _ = sender.send(value);
370                    } else {
371                        debug!(
372                            uuid = %uuid,
373                            len = value.len(),
374                            "unhandled notification, ignoring"
375                        );
376                    }
377                }
378            }
379            None => {
380                warn!("notification stream ended, device disconnected");
381                let _ = event_sender.send(ClockEvent::Disconnected);
382                is_authenticated.store(false, Ordering::SeqCst);
383
384                let device = ClockDevice {
385                    transport: transport.clone(),
386                    address,
387                    event_sender: event_sender.clone(),
388                    command_mutex: command_mutex.clone(),
389                    auth_token: auth_token.clone(),
390                    is_authenticated: is_authenticated.clone(),
391                    pending: pending.clone(),
392                    pending_data_response: pending_data_response.clone(),
393                    token_store: token_store.clone(),
394                };
395
396                match reconnect_and_restore(&device, 6).await {
397                    Ok(()) => {
398                        info!("reconnect successful, resuming notification task");
399                        continue;
400                    }
401                    Err(e) => {
402                        warn!("reconnect failed after all attempts: {e:?}");
403                        break;
404                    }
405                }
406            }
407        }
408    }
409}

411/// Reconnect with exponential backoff and full state recovery.
412///
413/// Delay sequence: 1s, 2s, 4s, 8s, 16s, 32s (capped).
414///
415/// After a successful BLE connect, re-subscribes to all GATT notify
416/// characteristics and re-authenticates with the stored token. This
417/// ensures the device is fully operational before commands resume.
418async fn reconnect_and_restore(device: &ClockDevice, max_attempts: u32) -> Result<()> {
419    let mut delay = Duration::from_secs(1);
420    let max_delay = Duration::from_secs(32);
421
422    for attempt in 1..=max_attempts {
423        debug!(attempt, delay_ms = delay.as_millis(), "attempting reconnect");
424
425        // Step 1: BLE Reconnect
426        if device.transport.connect(&device.address).await.is_err() {
427            tokio::time::sleep(delay).await;
428            delay = (delay * 2).min(max_delay);
429            continue;
430        }
431
432        // Step 2: Re-subscribe to all notify characteristics
433        let characteristics = [CharacteristicUuid::AuthNotify, CharacteristicUuid::DataNotify, CharacteristicUuid::SensorNotify];
434
435        let mut all_subscribed = true;
436        for char_uuid in &characteristics {
437            if device.transport.subscribe(*char_uuid).await.is_err() {
438                all_subscribed = false;
439            }
440        }
441
442        if !all_subscribed {
443            warn!("reconnect: GATT re-subscription failed, retrying");
444            tokio::time::sleep(delay).await;
445            delay = (delay * 2).min(max_delay);
446            continue;
447        }
448
449        // Step 3: Re-authenticate with stored token
450        let token = device.auth_token.lock().await;
451        if let Some(ref token) = *token
452            && device.authenticate(token).await.is_err()
453        {
454            warn!("reconnect: re-authentication failed, retrying");
455            tokio::time::sleep(delay).await;
456            delay = (delay * 2).min(max_delay);
457            continue;
458        }
459
460        // Step 4: Mark as authenticated and notify subscribers
461        device.is_authenticated.store(true, Ordering::SeqCst);
462        let _ = device.event_sender.send(ClockEvent::Reconnected);
463        info!("reconnect: state recovery complete");
464        return Ok(());
465    }
466
467    Err(ClockError::Transport("reconnect with state recovery failed".into()))
468}
