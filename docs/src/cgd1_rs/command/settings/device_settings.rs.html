18/// Device settings for the CGD1 alarm clock.
19///
20/// Maps to the 18-byte settings payload documented in `docs/BLE.md` §6.3
21/// and cross-referenced against the clOwOck Android implementation.
22/// The full frame is `13 01 [18 bytes payload]` for write and
23/// `13 02 [18 bytes payload]` for read response.
24///
25/// Invariants are enforced by construction via the newtype fields:
26/// - `volume` is in range 1–5 (validated by [`Volume`])
27/// - `brightness` and `night_brightness` are 0–150 in multiples of 10
28/// - `night_start`/`night_end` hours are 0–23, minutes 0–59
29/// - `timezone` is within -720 to +840 minutes
30/// - `screen_light_duration` is in range 0–255 seconds
31#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
32pub struct DeviceSettings {
33    /// Sound volume (1–5).
34    volume: Volume,
35    /// Flags (language, time format, temperature unit, master alarm disable).
36    #[serde(flatten)]
37    flags: SettingsFlags,
38    /// Timezone offset.
39    timezone: Timezone,
40    /// Screen light duration.
41    screen_light_duration: ScreenLightDuration,
42    /// Daytime brightness (0–150, multiple of 10).
43    brightness: Brightness,
44    /// Nighttime brightness (0–150, multiple of 10).
45    night_brightness: Brightness,
46    /// Night mode start time.
47    night_start: ClockTime,
48    /// Night mode end time.
49    night_end: ClockTime,
50    /// Whether night mode is enabled.
51    night_mode_enabled: bool,
52    /// Ringtone signature (4 bytes, `0xFF` when unused).
53    ringtone_signature: RingtoneSignature,
54}

201    /// Encode settings into the 18-byte payload for the settings frame.
202    ///
203    /// Layout (offsets within the payload, after the `[length] [command]` header):
204    /// ```text
205    /// [0]  Volume
206    /// [1]  Hdr1 (0x58)
207    /// [2]  Hdr2 (0x02)
208    /// [3]  Flags (language | time_format | temp_unit | master_alarm_disable)
209    /// [4]  Timezone (6-minute units)
210    /// [5]  Screen duration (seconds)
211    /// [6]  Packed brightness (high nibble = day, low nibble = night)
212    /// [7]  Night start hour
213    /// [8]  Night start minute
214    /// [9]  Night end hour
215    /// [10] Night end minute
216    /// [11] Timezone sign (0x01 = positive, 0x00 = negative)
217    /// [12] Night mode enabled (0x01 = enabled, 0x00 = disabled)
218    /// [13] Reserved (0xFF)
219    /// [14–17] Ringtone signature (4 bytes)
220    /// ```
221    ///
222    /// When night mode is disabled, the night start/end values are overwritten
223    /// to `00:00–00:01` in the encoded payload. This matches the firmware
224    /// workaround used by clOwOck and the official app, since the device does
225    /// not properly support disabling night mode via the enabled flag alone.
226    pub fn encode(&self) -> [u8; 18] {
227        let mut payload = [0u8; 18];
228        payload[0] = self.volume.value();
229        payload[1] = Self::HDR_BYTE_1;
230        payload[2] = Self::HDR_BYTE_2;
231        payload[3] = self.flags.byte();
232        payload[4] = self.timezone.encoded_units();
233        payload[5] = self.screen_light_duration.seconds();
234        payload[6] = (self.brightness.nibble() << 4) | self.night_brightness.nibble();
235        if self.night_mode_enabled {
236            payload[7] = self.night_start.hour();
237            payload[8] = self.night_start.minute();
238            payload[9] = self.night_end.hour();
239            payload[10] = self.night_end.minute();
240        } else {
241            // Firmware workaround: 1-minute night mode to effectively disable it.
242            payload[7] = 0;
243            payload[8] = 0;
244            payload[9] = 0;
245            payload[10] = 1;
246        }
247        payload[11] = self.timezone.sign_byte();
248        payload[12] = if self.night_mode_enabled { 0x01 } else { 0x00 };
249        payload[13] = 0xFF;
250        payload[14..18].copy_from_slice(&self.ringtone_signature.bytes());
251        payload
252    }
253
254    /// Decode settings from a raw payload read from the device.
255    ///
256    /// The payload should be the 18 bytes following the `[length] [command]` header.
257    pub fn decode(payload: &[u8]) -> Result<Self> {
258        if payload.len() < 18 {
259            return Err(ClockError::Parse("settings payload too short".into()));
260        }
261
262        let volume = Volume::new(payload[0])?;
263        let flags = payload[3];
264        let timezone = Timezone::from_encoded(payload[4], payload[11])?;
265        let screen_light_duration = ScreenLightDuration::new(payload[5])?;
266        let day_nibble = (payload[6] >> 4) & 0x0F;
267        let night_nibble = payload[6] & 0x0F;
268        let brightness = Brightness::from_nibble(day_nibble)?;
269        let night_brightness = Brightness::from_nibble(night_nibble)?;
270        let night_start = ClockTime::new(payload[7], payload[8])?;
271        let night_end = ClockTime::new(payload[9], payload[10])?;
272        let night_mode_enabled = payload[12] != 0x00;
273        let mut sig_bytes = [0u8; 4];
274        sig_bytes.copy_from_slice(&payload[14..18]);
275        let ringtone_signature = RingtoneSignature::from_bytes(sig_bytes);
276
277        let settings_flags = SettingsFlags::from_byte(flags);
278
279        Self::new(
280            volume,
281            settings_flags.time_format(),
282            settings_flags.temperature_unit(),
283            settings_flags.language(),
284            timezone,
285            screen_light_duration,
286            brightness,
287            night_brightness,
288            night_start,
289            night_end,
290            night_mode_enabled,
291            settings_flags.master_alarm_disabled(),
292            ringtone_signature,
293        )
294    }

328    #[test]
329    fn encode_known_values() {
330        let settings = sample_settings();
331        let payload = settings.encode();
332        assert_eq!(payload[0], 3); // volume
333        assert_eq!(payload[1], 0x58); // hdr1
334        assert_eq!(payload[2], 0x02); // hdr2
335        assert_eq!(payload[3], 0x11); // flags: English | master_alarm_disable
336        assert_eq!(payload[4], 10); // timezone: 60 min / 6 = 10
337        assert_eq!(payload[5], 10); // screen duration
338        assert_eq!(payload[6], 0x83); // brightness: day=8, night=3
339        assert_eq!(payload[7], 22); // night start hour
340        assert_eq!(payload[8], 0); // night start minute
341        assert_eq!(payload[9], 7); // night end hour
342        assert_eq!(payload[10], 0); // night end minute
343        assert_eq!(payload[11], 0x01); // timezone sign: positive
344        assert_eq!(payload[12], 0x01); // night mode enabled
345        assert_eq!(payload[13], 0xFF); // reserved
346        assert_eq!(&payload[14..18], &[0xFF, 0xFF, 0xFF, 0xFF]); // signature
347    }
