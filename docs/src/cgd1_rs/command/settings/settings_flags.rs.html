5use serde::Deserialize;
6use serde::Serialize;

8/// Packed flags for device settings, decoded into logical fields.
9///
10/// Encodes four values into a single byte on the wire:
11/// - Bit 0: `Language` (0 = Chinese, 1 = English)
12/// - Bit 1: `TimeFormat` (0 = 24h, 1 = 12h)
13/// - Bit 2: `TemperatureUnit` (0 = Celsius, 1 = Fahrenheit)
14/// - Bit 4: `master_alarm_disabled` (0 = enabled, 1 = disabled)
15#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
16pub struct SettingsFlags {
17    /// Display language.
18    language: Language,
19    /// Time display format.
20    time_format: TimeFormat,
21    /// Temperature display unit.
22    temperature_unit: TemperatureUnit,
23    /// Whether the master alarm switch is disabled.
24    master_alarm_disabled: bool,
25}
26
27impl SettingsFlags {
28    /// Flag bit for master alarm disable.
29    pub const FLAG_MASTER_ALARM_DISABLE: u8 = 0x10;
30
31    /// Create flags from the individual logical fields.
32    pub const fn new(language: Language, time_format: TimeFormat, temperature_unit: TemperatureUnit, master_alarm_disabled: bool) -> Self {
33        Self {
34            language,
35            time_format,
36            temperature_unit,
37            master_alarm_disabled,
38        }
39    }
40
41    /// Decode flags from a raw byte.
42    pub fn from_byte(byte: u8) -> Self {
43        Self {
44            language: Language::from_flags(byte),
45            time_format: TimeFormat::from_flags(byte),
46            temperature_unit: TemperatureUnit::from_flags(byte),
47            master_alarm_disabled: (byte & Self::FLAG_MASTER_ALARM_DISABLE) != 0,
48        }
49    }
50
51    /// Encode flags into the raw byte for the BLE payload.
52    pub fn byte(self) -> u8 {
53        self.language.flag_bit()
54            | self.time_format.flag_bit()
55            | self.temperature_unit.flag_bit()
56            | if self.master_alarm_disabled { Self::FLAG_MASTER_ALARM_DISABLE } else { 0 }
57    }
58
59    /// Get the language.
60    pub const fn language(self) -> Language {
61        self.language
62    }
63
64    /// Get the time format.
65    pub const fn time_format(self) -> TimeFormat {
66        self.time_format
67    }
68
69    /// Get the temperature unit.
70    pub const fn temperature_unit(self) -> TemperatureUnit {
71        self.temperature_unit
72    }
73
74    /// Whether the master alarm switch is disabled.
75    pub const fn master_alarm_disabled(self) -> bool {
76        self.master_alarm_disabled
77    }
78}
