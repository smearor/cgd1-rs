1use uuid::Uuid;
2
3/// BLE base UUID bytes for 16-bit UUID expansion.
4const BLE_BASE_UUID: [u8; 8] = [0x80, 0x00, 0x00, 0x80, 0x5f, 0x9b, 0x34, 0xfb];
5
6/// CGD1 GATT characteristic identifiers.
7#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
8pub enum CharacteristicUuid {
9    /// Auth Write - `00000001-0000-1000-8000-00805f9b34fb`
10    AuthWrite,
11    /// Auth Notify - `00000002-0000-1000-8000-00805f9b34fb`
12    AuthNotify,
13    /// Data Write - `0000000b-0000-1000-8000-00805f9b34fb`
14    DataWrite,
15    /// Data Notify - `0000000c-0000-1000-8000-00805f9b34fb`
16    DataNotify,
17    /// Sensor Notify - `00000100-0000-1000-8000-00805f9b34fb`
18    SensorNotify,
19    /// Battery Level - `0x2a19` (standard GATT)
20    BatteryLevel,
21}
22
23impl CharacteristicUuid {
24    /// Convert to the full 128-bit UUID.
25    pub fn uuid(self) -> Uuid {
26        match self {
27            Self::AuthWrite => Uuid::from_fields(0x00000001, 0x0000, 0x1000, &BLE_BASE_UUID),
28            Self::AuthNotify => Uuid::from_fields(0x00000002, 0x0000, 0x1000, &BLE_BASE_UUID),
29            Self::DataWrite => Uuid::from_fields(0x0000000b, 0x0000, 0x1000, &BLE_BASE_UUID),
30            Self::DataNotify => Uuid::from_fields(0x0000000c, 0x0000, 0x1000, &BLE_BASE_UUID),
31            Self::SensorNotify => Uuid::from_fields(0x00000100, 0x0000, 0x1000, &BLE_BASE_UUID),
32            Self::BatteryLevel => Uuid::from_fields(0x00002a19, 0x0000, 0x1000, &BLE_BASE_UUID),
33        }
34    }
35}
