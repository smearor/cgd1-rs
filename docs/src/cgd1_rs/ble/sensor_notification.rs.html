1use crate::error::ClockError;
2use crate::error::Result;
3use crate::types::Humidity;
4use crate::types::Temperature;

6/// Parsed sensor notification from the Sensor Notify characteristic.
7///
8/// Format: `[0x00] [TempLo] [TempHi] [HumLo] [HumHi]` (5 bytes).
9/// Temperature is a signed 16-bit little-endian value, scaled by / 100.
10/// Humidity is an unsigned 16-bit little-endian value, scaled by / 100.
11#[derive(Debug, Clone, PartialEq)]
12pub struct SensorNotification {
13    /// Temperature in degrees Celsius.
14    pub temperature: Temperature,
15    /// Relative humidity in percent.
16    pub humidity: Humidity,
17}
18
19impl SensorNotification {
20    /// Parse a raw sensor notification payload.
21    ///
22    /// The first byte must be `0x00` (sensor data header). Temperature and
23    /// humidity follow as little-endian 16-bit values, each divided by 100.
24    pub fn parse(payload: &[u8]) -> Result<Self> {
25        if payload.len() < 5 {
26            return Err(ClockError::Parse("sensor notification too short".to_string()));
27        }
28
29        if payload[0] != 0x00 {
30            return Err(ClockError::Parse(format!("unexpected sensor header byte: 0x{:02X}", payload[0])));
31        }
32
33        let temp_raw = i16::from_le_bytes([payload[1], payload[2]]);
34        let humidity_raw = u16::from_le_bytes([payload[3], payload[4]]);
35
36        Ok(Self {
37            temperature: Temperature::new(temp_raw as f32 / 100.0),
38            humidity: Humidity::new(humidity_raw as f32 / 100.0),
39        })
40    }
41}

43#[cfg(test)]
44mod tests {
45    use super::*;
46
47    #[test]
48    fn parse_valid_notification() {
49        // [0x00] [TempLo] [TempHi] [HumLo] [HumHi]
50        // Temp = 2345 → 23.45 °C, Hum = 5600 → 56.00 %
51        let payload = [0x00, 0x29, 0x09, 0xE0, 0x15];
52        let sensor = SensorNotification::parse(&payload).unwrap();
53        assert_eq!(sensor.temperature.value(), 23.45);
54        assert_eq!(sensor.humidity.value(), 56.0);
55    }
56
57    #[test]
58    fn parse_negative_temperature() {
59        // Temp = -550 → -5.50 °C, LE bytes: [0xDA, 0xFD]
60        let payload = [0x00, 0xDA, 0xFD, 0x64, 0x00];
61        let sensor = SensorNotification::parse(&payload).unwrap();
62        assert_eq!(sensor.temperature.value(), -5.5);
63        assert_eq!(sensor.humidity.value(), 1.0);
64    }
65
66    #[test]
67    fn parse_zero_values() {
68        let payload = [0x00, 0x00, 0x00, 0x00, 0x00];
69        let sensor = SensorNotification::parse(&payload).unwrap();
70        assert_eq!(sensor.temperature.value(), 0.0);
71        assert_eq!(sensor.humidity.value(), 0.0);
72    }
73
74    #[test]
75    fn parse_too_short() {
76        let payload = [0x00, 0x29, 0x09];
77        assert!(SensorNotification::parse(&payload).is_err());
78    }
79
80    #[test]
81    fn parse_wrong_header() {
82        let payload = [0x01, 0x29, 0x09, 0xE0, 0x15];
83        assert!(SensorNotification::parse(&payload).is_err());
84    }
85}
