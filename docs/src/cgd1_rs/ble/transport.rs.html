11/// BLE transport abstraction for CGD1 communication.
12///
13/// This trait keeps the door open for alternative backends (e.g., `bluer`).
14/// The implementation manages a single active BLE connection.
15#[async_trait]
16pub trait BleTransport: Send + Sync {
17    /// Start scanning for devices with the given service-data UUID filter.
18    async fn start_scan(&self, filter_uuid: Uuid) -> Result<()>;
19
20    /// Stop an active scan.
21    async fn stop_scan(&self) -> Result<()>;
22
23    /// Receive the next advertisement event during scanning.
24    async fn next_advertisement(&self) -> Option<AdvertisementData>;
25
26    /// Connect to a peripheral by MAC address.
27    async fn connect(&self, address: &MacAddress) -> Result<()>;
28
29    /// Disconnect from the current peripheral.
30    async fn disconnect(&self) -> Result<()>;
31
32    /// Write data to a characteristic.
33    async fn write(&self, characteristic: CharacteristicUuid, data: &[u8]) -> Result<()>;
34
35    /// Write a command frame to the characteristic associated with the command.
36    ///
37    /// This is a convenience method that derives the correct GATT characteristic
38    /// from the [`Command`] variant, eliminating the risk of sending a command
39    /// on the wrong characteristic.
40    async fn write_command(&self, command: Command, data: &[u8]) -> Result<()> {
41        self.write(command.characteristic(), data).await
42    }
43
44    /// Encode and write a command frame to the appropriate characteristic.
45    ///
46    /// Builds a [`CommandFrame`] from the given [`Command`] and payload,
47    /// encodes it as `[length] [command] [payload...]`, and sends it to
48    /// the characteristic associated with the command.
49    async fn write_frame(&self, command: Command, payload: &[u8]) -> Result<()> {
50        let frame = CommandFrame::from_command(command, payload.to_vec());
51        let encoded = frame.encode();
52        self.write(command.characteristic(), &encoded).await
53    }
54
55    /// Subscribe to notifications from a characteristic.
56    async fn subscribe(&self, characteristic: CharacteristicUuid) -> Result<()>;
57
58    /// Receive the next notification value.
59    ///
60    /// Returns the raw value bytes and the characteristic UUID that produced
61    /// the notification, enabling the notification task to route frames
62    /// correctly.
63    async fn next_notification(&self) -> Option<(Uuid, Vec<u8>)>;
64
65    /// Read a characteristic value.
66    async fn read(&self, characteristic: CharacteristicUuid) -> Result<Vec<u8>>;
67
68    /// Request a larger ATT MTU via the BLE MTU exchange procedure.
69    ///
70    /// The default ATT MTU is 23 bytes (20 bytes usable payload). Audio
71    /// data packets require 130 bytes, so an MTU exchange must succeed
72    /// before uploading ringtones. Returns the negotiated MTU on success.
73    async fn request_mtu(&self, mtu: u16) -> Result<u16>;
74
75    /// Check if the device is currently connected.
76    fn is_connected(&self) -> bool;
77}
