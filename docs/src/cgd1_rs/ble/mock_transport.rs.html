16/// In-memory mock BLE transport for testing.
17///
18/// Captures written data so tests can assert on frame content, and allows
19/// injecting notification values to simulate device responses.
20/// When a command frame is written, an ACK is automatically generated on the
21/// corresponding notify characteristic.
22pub struct MockBleTransport {
23    connected: AtomicBool,
24    writes: Mutex<Vec<(CharacteristicUuid, Vec<u8>)>>,
25    notifications_tx: mpsc::UnboundedSender<(Uuid, Vec<u8>)>,
26    notifications_rx: Mutex<mpsc::UnboundedReceiver<(Uuid, Vec<u8>)>>,
27    advertisements: Mutex<Vec<AdvertisementData>>,
28    subscribed: Mutex<Vec<CharacteristicUuid>>,
29    read_values: Mutex<Vec<(CharacteristicUuid, Vec<u8>)>>,
30    mtu: Mutex<u16>,
31    auto_ack: AtomicBool,
32}
33
34impl MockBleTransport {
35    /// Create a new mock transport with auto-ACK enabled.
36    pub fn new() -> Self {
37        let (tx, rx) = mpsc::unbounded_channel();
38        Self {
39            connected: AtomicBool::new(false),
40            writes: Mutex::new(Vec::new()),
41            notifications_tx: tx,
42            notifications_rx: Mutex::new(rx),
43            advertisements: Mutex::new(Vec::new()),
44            subscribed: Mutex::new(Vec::new()),
45            read_values: Mutex::new(Vec::new()),
46            mtu: Mutex::new(23),
47            auto_ack: AtomicBool::new(true),
48        }
49    }
50
51    /// Push a notification that will be returned by `next_notification`.
52    pub fn push_notification(&self, uuid: Uuid, data: Vec<u8>) {
53        let _ = self.notifications_tx.send((uuid, data));
54    }
55
56    /// Push an advertisement that will be returned by `next_advertisement`.
57    pub async fn push_advertisement(&self, adv: AdvertisementData) {
58        self.advertisements.lock().await.push(adv);
59    }
60
61    /// Set the value returned by `read` for a given characteristic.
62    pub async fn set_read_value(&self, characteristic: CharacteristicUuid, data: Vec<u8>) {
63        self.read_values.lock().await.push((characteristic, data));
64    }
65
66    /// Drain and return all captured writes.
67    pub async fn drain_writes(&self) -> Vec<(CharacteristicUuid, Vec<u8>)> {
68        self.writes.lock().await.drain(..).collect()
69    }
70
71    /// Get the list of subscribed characteristics.
72    pub async fn subscribed(&self) -> Vec<CharacteristicUuid> {
73        self.subscribed.lock().await.clone()
74    }
75
76    /// Enable or disable auto-ACK generation on writes.
77    pub fn set_auto_ack(&self, enabled: bool) {
78        self.auto_ack.store(enabled, Ordering::SeqCst);
79    }
80
81    /// Determine the notify characteristic for a given write characteristic.
82    fn notify_for(characteristic: CharacteristicUuid) -> Option<CharacteristicUuid> {
83        match characteristic {
84            CharacteristicUuid::AuthWrite => Some(CharacteristicUuid::AuthNotify),
85            CharacteristicUuid::DataWrite => Some(CharacteristicUuid::DataNotify),
86            _ => None,
87        }
88    }
89
90    /// Auto-generate an ACK for a command frame if auto-ACK is enabled.
91    ///
92    /// Frame format: `[length] [command_byte] [payload...]`
93    /// ACK format: `[0x04] [0xff] [command_byte] [0x00] [0x00]`
94    fn maybe_auto_ack(&self, characteristic: CharacteristicUuid, data: &[u8]) {
95        if !self.auto_ack.load(Ordering::SeqCst) {
96            return;
97        }
98        if data.len() < 2 {
99            return;
100        }
101        let Some(notify) = Self::notify_for(characteristic) else {
102            return;
103        };
104        let command_byte = data[1];
105        let ack = vec![0x04, 0xff, command_byte, 0x00, 0x00];
106        self.push_notification(notify.uuid(), ack);
107    }
108}
109
110impl Default for MockBleTransport {
111    fn default() -> Self {
112        Self::new()
113    }
114}

116#[async_trait]
117impl BleTransport for MockBleTransport {
118    async fn start_scan(&self, _filter_uuid: Uuid) -> Result<()> {
119        Ok(())
120    }
121
122    async fn stop_scan(&self) -> Result<()> {
123        Ok(())
124    }
125
126    async fn next_advertisement(&self) -> Option<AdvertisementData> {
127        self.advertisements.lock().await.pop()
128    }
129
130    async fn connect(&self, _address: &MacAddress) -> Result<()> {
131        self.connected.store(true, Ordering::SeqCst);
132        Ok(())
133    }
134
135    async fn disconnect(&self) -> Result<()> {
136        self.connected.store(false, Ordering::SeqCst);
137        Ok(())
138    }
139
140    async fn write(&self, characteristic: CharacteristicUuid, data: &[u8]) -> Result<()> {
141        self.writes.lock().await.push((characteristic, data.to_vec()));
142        self.maybe_auto_ack(characteristic, data);
143        Ok(())
144    }
145
146    async fn subscribe(&self, characteristic: CharacteristicUuid) -> Result<()> {
147        self.subscribed.lock().await.push(characteristic);
148        Ok(())
149    }
150
151    async fn next_notification(&self) -> Option<(Uuid, Vec<u8>)> {
152        self.notifications_rx.lock().await.recv().await
153    }
154
155    async fn read(&self, characteristic: CharacteristicUuid) -> Result<Vec<u8>> {
156        let mut read_values = self.read_values.lock().await;
157        let pos = read_values.iter().position(|(c, _)| *c == characteristic);
158        if let Some(idx) = pos {
159            Ok(read_values.remove(idx).1)
160        } else {
161            Err(ClockError::Transport("no read value set for characteristic".into()))
162        }
163    }
164
165    async fn request_mtu(&self, mtu: u16) -> Result<u16> {
166        let mut current = self.mtu.lock().await;
167        *current = mtu;
168        Ok(mtu)
169    }
170
171    fn is_connected(&self) -> bool {
172        self.connected.load(Ordering::SeqCst)
173    }
174}

176#[cfg(test)]
177mod tests {
178    use super::*;
179    use crate::CharacteristicUuid;
180
181    #[tokio::test]
182    async fn mock_write_capture() {
183        let transport = MockBleTransport::new();
184        transport.set_auto_ack(false);
185        transport.write(CharacteristicUuid::AuthWrite, &[0x11, 0x01]).await.unwrap();
186        let writes = transport.drain_writes().await;
187        assert_eq!(writes.len(), 1);
188        assert_eq!(writes[0].0, CharacteristicUuid::AuthWrite);
189        assert_eq!(writes[0].1, vec![0x11, 0x01]);
190    }
191
192    #[tokio::test]
193    async fn mock_notification_push_pop() {
194        let transport = MockBleTransport::new();
195        let uuid = CharacteristicUuid::AuthNotify.uuid();
196        transport.push_notification(uuid, vec![0x04, 0xff, 0x01, 0x00, 0x00]);
197        let (recv_uuid, data) = transport.next_notification().await.unwrap();
198        assert_eq!(recv_uuid, uuid);
199        assert_eq!(data, vec![0x04, 0xff, 0x01, 0x00, 0x00]);
200    }
201
202    #[tokio::test]
203    async fn mock_connect_disconnect() {
204        let transport = MockBleTransport::new();
205        assert!(!transport.is_connected());
206        let addr = MacAddress::parse("AA:BB:CC:DD:EE:FF").unwrap();
207        transport.connect(&addr).await.unwrap();
208        assert!(transport.is_connected());
209        transport.disconnect().await.unwrap();
210        assert!(!transport.is_connected());
211    }
212
213    #[tokio::test]
214    async fn mock_subscribe_tracking() {
215        let transport = MockBleTransport::new();
216        transport.subscribe(CharacteristicUuid::AuthNotify).await.unwrap();
217        transport.subscribe(CharacteristicUuid::DataNotify).await.unwrap();
218        let subs = transport.subscribed().await;
219        assert_eq!(subs, vec![CharacteristicUuid::AuthNotify, CharacteristicUuid::DataNotify]);
220    }
221
222    #[tokio::test]
223    async fn mock_auto_ack() {
224        let transport = MockBleTransport::new();
225        transport
226            .write(CharacteristicUuid::DataWrite, &[0x07, 0x05, 0x02, 0x01, 0x07, 0x1E, 0x3E, 0x01])
227            .await
228            .unwrap();
229        let (uuid, data) = transport.next_notification().await.unwrap();
230        assert_eq!(uuid, CharacteristicUuid::DataNotify.uuid());
231        assert_eq!(data, vec![0x04, 0xff, 0x05, 0x00, 0x00]);
232    }
