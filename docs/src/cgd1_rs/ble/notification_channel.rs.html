9/// Pinned, boxed notification stream from btleplug.
10///
11/// This is the stream returned by `Peripheral::notifications()`, yielding
12/// `ValueNotification` items as GATT notifications arrive from the device.
13/// It is pinned because the underlying stream may be self-referential.
14pub(crate) type NotificationStream = Pin<Box<dyn Stream<Item = ValueNotification> + Send>>;
15
16/// Per-device notification channel: GATT value stream plus a disconnect signal.
17///
18/// Each connected device has one `NotificationChannel` stored in
19/// `BtleplugTransport::notification_streams`. The `stream` field yields
20/// incoming GATT notifications, while `disconnect_rx` is a watch channel
21/// receiver that is signalled when the device disconnects (either explicitly
22/// via `disconnect()` or by the event monitor detecting a `DeviceDisconnected`
23/// CentralEvent).
24///
25/// `next_notification()` uses `tokio::select!` to race between the stream
26/// and the disconnect signal, so that a silent BLE disconnect (where the
27/// notification stream never ends) is still detected promptly.
28pub(crate) struct NotificationChannel {
29    /// The GATT notification stream from btleplug.
30    ///
31    /// Yields `ValueNotification` items as the device sends data on subscribed
32    /// characteristics (AuthNotify, DataNotify, SensorNotify).
33    pub stream: NotificationStream,
34
35    /// Watch channel receiver for the disconnect signal.
36    ///
37    /// Holds `false` while connected. When the sender transmits `true`,
38    /// `next_notification()` returns `None`, causing the notification task
39    /// to begin reconnect logic.
40    pub disconnect_rx: watch::Receiver<bool>,
41}
42
43/// Shared, thread-safe handle to a [`NotificationChannel`].
44///
45/// Stored in `BtleplugTransport::notification_streams` keyed by MAC address.
46/// The `Arc<Mutex<...>>` allows the notification task to lock the channel
47/// while polling the stream, without holding the `notification_streams` map
48/// lock.
49pub(crate) type SharedNotificationChannel = Arc<Mutex<NotificationChannel>>;
