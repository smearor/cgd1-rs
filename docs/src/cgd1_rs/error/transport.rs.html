5/// BLE transport error details.
6///
7/// Replaces the former `Transport(String)` variant with structured
8/// error information for each known transport failure mode.
9#[derive(Debug, Clone, PartialEq, Eq, Error)]
10pub enum TransportError {
11    /// No Bluetooth adapter was found on the system.
12    #[error("no Bluetooth adapter found")]
13    NoAdapter,
14
15    /// The device with the given address was not found during scanning.
16    #[error("device not found: {address}")]
17    DeviceNotFound {
18        /// The MAC address that was searched for.
19        address: MacAddress,
20    },
21
22    /// The requested GATT characteristic was not found on the device.
23    #[error("characteristic not found: {characteristic}")]
24    CharacteristicNotFound {
25        /// The characteristic that was requested.
26        characteristic: CharacteristicUuid,
27    },
28
29    /// No read value was queued for the requested characteristic.
30    #[error("no read value set for characteristic")]
31    NoReadValue,
32
33    /// A pending request was canceled before a response arrived.
34    #[error("pending request canceled")]
35    RequestCanceled,
36
37    /// A response was canceled before it could be delivered.
38    #[error("{context} response canceled")]
39    ResponseCanceled {
40        /// What kind of response was canceled (e.g. "firmware", "settings").
41        context: String,
42    },
43
44    /// Reconnect with state recovery failed.
45    #[error("reconnect with state recovery failed")]
46    ReconnectFailed,
47
48    /// The device is not connected.
49    #[error("not connected")]
50    NotConnected,
51
52    /// The connected device MAC is unknown to the transport.
53    #[error("unknown device MAC: {mac}")]
54    UnknownDeviceMac {
55        /// The MAC address that was not recognized.
56        mac: MacAddress,
57    },
58
59    /// The transport does not support reading the given characteristic.
60    #[error("transport does not support read for characteristic {characteristic}")]
61    UnsupportedRead {
62        /// The characteristic that was requested.
63        characteristic: CharacteristicUuid,
64    },
65
66    /// System time error (e.g. clock went backwards).
67    #[error("system time error: {0}")]
68    SystemTime(String),
69
70    /// Catch-all for other transport errors with a custom message.
71    #[error("{0}")]
72    Other(String),
73}
