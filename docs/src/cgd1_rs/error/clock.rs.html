22/// Errors returned by the cgd1-rs library.
23#[derive(Debug, Error)]
24pub enum ClockError {
25    /// BLE transport error.
26    #[error(transparent)]
27    Transport(#[from] TransportError),
28
29    /// Authentication failed (token rejected by device).
30    #[error("{0}")]
31    AuthFailed(AuthFailedError),
32
33    /// No authentication token available for this device.
34    #[error("no auth token: device not paired")]
35    NoAuthToken,
36
37    /// Command was rejected by the device (non-success ACK status).
38    #[error("command rejected: command={command:#04x}, status={status}")]
39    CommandRejected {
40        /// The command byte that was rejected.
41        command: u8,
42        /// The ACK status from the device.
43        status: AckStatus,
44    },
45
46    /// Timeout waiting for a response from the device.
47    #[error("timeout waiting for response")]
48    Timeout,
49
50    /// Device is not connected.
51    #[error("not connected")]
52    NotConnected,
53
54    /// Device is already connected.
55    #[error("already connected")]
56    AlreadyConnected,
57
58    /// Invalid alarm slot index (must be 0-15).
59    #[error("invalid alarm slot: {0}")]
60    InvalidAlarmSlot(u8),
61
62    /// Invalid settings value (out of range).
63    #[error("invalid settings value: {0}")]
64    InvalidSettings(String),
65
66    /// Failed to parse an advertisement or notification.
67    #[error("parse error: {0}")]
68    Parse(String),
69
70    /// Failed to parse a day mask from a string.
71    #[error(transparent)]
72    DayMaskParse(#[from] DayMaskParseError),
73
74    /// Failed to parse an alarm slot index from a string.
75    #[error(transparent)]
76    AlarmSlotIndexParse(#[from] AlarmSlotIndexParseError),
77
78    /// Failed to parse a MAC address from a string.
79    #[error(transparent)]
80    MacAddressParse(#[from] MacAddressParseError),
81
82    /// Failed to parse a battery level from a string.
83    #[error(transparent)]
84    BatteryLevelParse(#[from] BatteryLevelParseError),
85
86    /// Failed to parse a humidity from a string.
87    #[error(transparent)]
88    HumidityParse(#[from] HumidityParseError),
89
90    /// Failed to parse a temperature from a string.
91    #[error(transparent)]
92    TemperatureParse(#[from] TemperatureParseError),
93
94    /// Failed to parse a brightness from a string.
95    #[error(transparent)]
96    BrightnessParse(#[from] BrightnessParseError),
97
98    /// Failed to parse a language from a string.
99    #[error(transparent)]
100    LanguageParse(#[from] LanguageParseError),
101
102    /// Failed to parse a time format from a string.
103    #[error(transparent)]
104    TimeFormatParse(#[from] TimeFormatParseError),
105
106    /// Failed to parse a timezone from a string.
107    #[error(transparent)]
108    TimezoneParse(#[from] TimezoneParseError),
109
110    /// Failed to parse a temperature unit from a string.
111    #[error(transparent)]
112    TemperatureUnitParse(#[from] TemperatureUnitParseError),
113
114    /// Failed to parse a volume from a string.
115    #[error(transparent)]
116    VolumeParse(#[from] VolumeParseError),
117
118    /// Failed to parse a clock time from a string.
119    #[error(transparent)]
120    ClockTimeParse(#[from] ClockTimeParseError),
121
122    /// Failed to parse a screen duration from a string.
123    #[error(transparent)]
124    ScreenLightDurationParse(#[from] ScreenLightDurationParseError),
125
126    /// Failed to parse a ringtone signature from a string.
127    #[error(transparent)]
128    RingtoneSignatureParse(#[from] RingtoneSignatureParseError),
129
130    /// I/O error.
131    #[error("IO error: {0}")]
132    Io(#[from] std::io::Error),
133
134    /// Internal btleplug error.
135    #[error("btleplug error: {0}")]
136    Btleplug(#[from] btleplug::Error),
137}
