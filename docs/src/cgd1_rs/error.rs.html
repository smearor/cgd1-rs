4/// Errors returned by the cgd1-rs library.
5#[derive(Debug, thiserror::Error)]
6pub enum ClockError {
7    /// BLE transport error.
8    #[error("BLE transport error: {0}")]
9    Transport(String),
10
11    /// Authentication failed (token rejected by device).
12    #[error("authentication failed: {0}")]
13    AuthFailed(String),
14
15    /// No authentication token available for this device.
16    #[error("no auth token: device not paired")]
17    NoAuthToken,
18
19    /// Command was rejected by the device (non-success ACK status).
20    #[error("command rejected: command={command:#04x}, status={status}")]
21    CommandRejected {
22        /// The command byte that was rejected.
23        command: u8,
24        /// The ACK status from the device.
25        status: crate::command::AckStatus,
26    },
27
28    /// Timeout waiting for a response from the device.
29    #[error("timeout waiting for response")]
30    Timeout,
31
32    /// Device is not connected.
33    #[error("not connected")]
34    NotConnected,
35
36    /// Device is already connected.
37    #[error("already connected")]
38    AlreadyConnected,
39
40    /// Invalid alarm slot index (must be 0-15).
41    #[error("invalid alarm slot: {0}")]
42    InvalidAlarmSlot(u8),
43
44    /// Invalid settings value (out of range).
45    #[error("invalid settings value: {0}")]
46    InvalidSettings(String),
47
48    /// Failed to parse an advertisement or notification.
49    #[error("parse error: {0}")]
50    Parse(String),
51
52    /// I/O error.
53    #[error("IO error: {0}")]
54    Io(#[from] std::io::Error),
55
56    /// Internal btleplug error.
57    #[error("btleplug error: {0}")]
58    Btleplug(#[from] btleplug::Error),
59}
