27/// Persistent REPL session state.
28struct ReplState {
29    transport: Arc<dyn BleTransport>,
30    manager: ClockManager,
31    token_store: Arc<FileTokenStore>,
32    connection: Option<DeviceConnection>,
33}
34
35impl ReplState {
36    /// Create a new REPL state with the given backend.
37    async fn new(backend: Backend) -> Result<Self, CliError> {
38        let transport = backend.create_transport().await?;
39        let manager = ClockManager::new(transport.clone());
40        let token_store = Arc::new(FileTokenStore::default_directory());
41        Ok(Self {
42            transport,
43            manager,
44            token_store,
45            connection: None,
46        })
47    }
48
49    /// Connect to a device by MAC address.
50    async fn connect(&mut self, mac: &MacAddress) -> Result<(), CliError> {
51        if self.connection.is_some() {
52            eprintln!("Already connected. Use `disconnect` first.");
53            return Ok(());
54        }
55
56        let device = self.manager.connect(mac).await?;
57        let token_result = self.token_store.load_or_generate(mac);
58        device.set_token_store(self.token_store.clone() as Arc<dyn TokenStore>).await;
59        device.authenticate(&token_result).await?;
60        self.connection = Some(DeviceConnection::new(self.transport.clone(), device));
61        println!("Connected to {mac}.");
62        Ok(())
63    }
64
65    /// Disconnect from the current device.
66    async fn disconnect(&mut self) -> Result<(), CliError> {
67        if let Some(conn) = self.connection.take() {
68            let mac = *conn.device().address();
69            self.manager.disconnect(&mac).await?;
70            println!("Disconnected from {mac}.");
71        } else {
72            println!("Not connected.");
73        }
74        Ok(())
75    }
76
77    /// Get the connected device, or print an error.
78    fn device(&self) -> Option<&ClockDevice> {
79        self.connection.as_ref().map(|c| c.device())
80    }
81
82    /// Run a scan and print results.
83    async fn scan(&self, duration: ScanDuration) -> Result<(), CliError> {
84        let scanner = ClockScanner::new(self.transport.clone());
85        let devices = scanner.scan_active(std::time::Duration::from(duration)).await?;
86
87        if devices.is_empty() {
88            println!("No CGD1 devices found.");
89            return Ok(());
90        }
91
92        println!("Found {} device(s):", devices.len());
93        for device in &devices {
94            println!("  MAC: {}", device.address);
95            if let Some(ad) = &device.advertisement {
96                println!("    Temperature: {:.1} C", ad.temperature.value());
97                println!("    Humidity: {:.1} %", ad.humidity.value());
98                println!("    Battery: {} %", ad.battery.value());
99            }
100        }
101        Ok(())
102    }
103}

105/// Run the REPL loop.
106pub async fn run(backend: Backend, initial_address: Option<MacAddress>) -> Result<(), CliError> {
107    let mut state = ReplState::new(backend).await?;
108    let mut rl = DefaultEditor::new().expect("failed to initialize readline");
109
110    if let Some(mac) = initial_address {
111        state.connect(&mac).await?;
112    }
113
114    println!("cgd1 REPL — type 'help' for commands, 'exit' to quit.");
115
116    loop {
117        let prompt = match state.connection {
118            Some(ref conn) => format!("cgd1({})> ", conn.device().address()),
119            None => "cgd1> ".to_string(),
120        };
121
122        let line = match rl.readline(&prompt) {
123            Ok(line) => {
124                let trimmed = line.trim();
125                if trimmed.is_empty() {
126                    continue;
127                }
128                let _ = rl.add_history_entry(trimmed);
129                trimmed.to_string()
130            }
131            Err(ReadlineError::Interrupted | ReadlineError::Eof) => {
132                println!("bye.");
133                break;
134            }
135            Err(e) => {
136                eprintln!("readline error: {e}");
137                break;
138            }
139        };
140
141        if let Err(e) = handle_line(&mut state, &line).await {
142            eprintln!("error: {e}");
143        }
144    }
145
146    if let Some(conn) = state.connection.take() {
147        let mac = *conn.device().address();
148        let _ = state.manager.disconnect(&mac).await;
149    }
150
151    Ok(())
152}

154/// Handle a single REPL input line.
155async fn handle_line(state: &mut ReplState, line: &str) -> Result<(), CliError> {
156    let tokens = shell_words::split(line).unwrap_or_default();
157    if tokens.is_empty() {
158        return Ok(());
159    }
160
161    match tokens[0].as_str() {
162        "exit" | "quit" => {
163            std::process::exit(0);
164        }
165        "help" => {
166            print_help();
167            return Ok(());
168        }
169        "connect" => {
170            let mac: MacAddress = tokens
171                .get(1)
172                .ok_or_else(|| CliError::Core(ClockError::Parse("missing MAC address".into())))?
173                .parse()
174                .map_err(|e| CliError::Core(ClockError::Parse(format!("invalid MAC: {e}"))))?;
175            return state.connect(&mac).await;
176        }
177        "disconnect" => {
178            return state.disconnect().await;
179        }
180        _ => {}
181    }
182
183    // Parse as a full CLI command (reuses clap definitions).
184    // If connected and the command requires <ADDRESS> but the user omitted it,
185    // inject the connected device's address automatically.
186    let tokens = inject_address_if_needed(state, &tokens);
187    let mut args = vec!["cgd1".to_string()];
188    args.extend(tokens.iter().cloned());
189
190    let cli = match Cli::try_parse_from(&args) {
191        Ok(cli) => cli,
192        Err(e) => {
193            e.print().ok();
194            return Ok(());
195        }
196    };
197
198    dispatch(state, cli).await
199}

201/// Commands that take <ADDRESS> as their first positional argument.
202const ADDRESS_COMMANDS: &[&str] = &[
203    "sync-time",
204    "alarm-list",
205    "alarm-set",
206    "alarm-delete",
207    "settings-read",
208    "settings-write",
209    "brightness",
210    "ringtone-preview",
211    "ringtone-upload",
212    "firmware",
213    "battery",
214    "monitor",
215];
216
217/// If the user is connected and typed a device command without a MAC address,
218/// inject the connected address as the second token.
219fn inject_address_if_needed(state: &ReplState, tokens: &[String]) -> Vec<String> {
220    if tokens.is_empty() {
221        return tokens.to_vec();
222    }
223
224    let command = tokens[0].as_str();
225    if !ADDRESS_COMMANDS.contains(&command) {
226        return tokens.to_vec();
227    }
228
229    // If the second token looks like a MAC (contains ':'), the user provided it.
230    if tokens.get(1).is_some_and(|t| t.contains(':')) {
231        return tokens.to_vec();
232    }
233
234    // If connected, inject the address after the command name.
235    if let Some(ref conn) = state.connection {
236        let mut result = vec![tokens[0].clone()];
237        result.push(conn.device().address().to_string());
238        result.extend(tokens.iter().skip(1).cloned());
239        return result;
240    }
241
242    tokens.to_vec()
243}

245/// Dispatch a parsed CLI command in the REPL context.
246async fn dispatch(state: &mut ReplState, cli: Cli) -> Result<(), CliError> {
247    match cli.command {
248        Commands::Scan { duration } => {
249            state.scan(duration).await?;
250        }
251        Commands::Repl { .. } => {
252            eprintln!("nested REPL is not supported.");
253        }
254        Commands::SyncTime { address } => {
255            let device = require_device(state, &address)?;
256            device.sync_time_now().await?;
257            let token_result = state.token_store.load_or_generate(&address);
258            if token_result.is_new() {
259                state.token_store.save(&address, &token_result)?;
260                info!(address = %address, "new token persisted after successful sync_time");
261            }
262            println!("Time synchronized for {address}.");
263        }
264        Commands::Firmware { address } => {
265            let device = require_device(state, &address)?;
266            let version = device.read_firmware().await?;
267            println!("Firmware: {version}");
268        }
269        Commands::Battery { address } => {
270            let device = require_device(state, &address)?;
271            let level = device.read_battery().await?;
272            println!("Battery: {} %", level.value());
273        }
274        Commands::SettingsRead { address } => {
275            let device = require_device(state, &address)?;
276            let settings = device.read_settings().await?;
277            println!("Volume: {}", settings.volume());
278            println!("Brightness: {}", settings.brightness().nibble() * 10);
279            println!("Night brightness: {}", settings.night_brightness().nibble() * 10);
280            println!("Time format: {}", if settings.time_format() == TimeFormat::TwelveHour { "12h" } else { "24h" });
281            println!(
282                "Temperature unit: {}",
283                if settings.temperature_unit() == TemperatureUnit::Fahrenheit {
284                    "F"
285                } else {
286                    "C"
287                }
288            );
289            println!("Language: {}", if settings.language() == Language::English { "en" } else { "zh" });
290            println!("Timezone: {:+} min", settings.timezone().minutes());
291            println!("Screen light duration: {}", settings.screen_light_duration());
292            println!(
293                "Night mode: {} ({}–{})",
294                if settings.night_mode_enabled() { "on" } else { "off" },
295                settings.night_start(),
296                settings.night_end(),
297            );
298            println!("Ringtone signature: {}", settings.ringtone_signature());
299        }
300        Commands::SettingsWrite {
301            address,
302            volume,
303            brightness,
304            night_brightness,
305            timezone,
306            time_format,
307            temp_unit,
308            language,
309        } => {
310            let device = require_device(state, &address)?;
311            let mut settings = device.read_settings().await?;
312            if let Some(vol) = volume {
313                settings = settings.with_volume(vol)?;
314            }
315            if let Some(b) = brightness {
316                settings = settings.with_brightness(b)?;
317            }
318            if let Some(nb) = night_brightness {
319                settings = settings.with_night_brightness(nb)?;
320            }
321            if let Some(tz) = timezone {
322                settings = settings.with_timezone(tz)?;
323            }
324            if let Some(tf) = time_format {
325                settings = settings.with_time_format(tf);
326            }
327            if let Some(tu) = temp_unit {
328                settings = settings.with_temperature_unit(tu);
329            }
330            if let Some(lang) = language {
331                settings = settings.with_language(lang);
332            }
333            device.write_settings(&settings).await?;
334            println!("Settings updated.");
335        }
336        Commands::AlarmList { address } => {
337            let device = require_device(state, &address)?;
338            let slots = device.read_alarms().await?;
339            if slots.is_empty() {
340                println!("No alarms set.");
341                return Ok(());
342            }
343            println!("{:<6} {:<8} {:<3} {:<3} {:<6} {:<6}", "Slot", "Enabled", "HH", "MM", "Days", "Snooze");
344            for slot in &slots {
345                println!(
346                    "{:<6} {:<8} {:02}   {:02}   0x{:02x}  {}",
347                    slot.index.value(),
348                    if slot.entry.enabled() { "yes" } else { "no" },
349                    slot.entry.hour(),
350                    slot.entry.minute(),
351                    slot.entry.repeat_mask().value(),
352                    if slot.entry.snooze() { "yes" } else { "no" },
353                );
354            }
355        }
356        Commands::AlarmSet {
357            address,
358            slot,
359            time,
360            repeat,
361            no_snooze,
362        } => {
363            let device = require_device(state, &address)?;
364            let entry = AlarmEntry::new(time, repeat, true, !no_snooze);
365            device.set_alarm(&entry, slot).await?;
366            println!("Alarm set at slot {}.", slot.value());
367        }
368        Commands::AlarmDelete { address, slot } => {
369            let device = require_device(state, &address)?;
370            device.delete_alarm(slot).await?;
371            println!("Alarm deleted at slot {}.", slot.value());
372        }
373        Commands::Brightness { address, value } => {
374            let device = require_device(state, &address)?;
375            device.set_brightness(value).await?;
376            println!("Brightness set to {}.", value);
377        }
378        Commands::RingtonePreview { address, volume } => {
379            let device = require_device(state, &address)?;
380            device.preview_ringtone(volume).await?;
381            println!("Ringtone preview started.");
382        }
383        Commands::RingtoneUpload { address, file, signature } => {
384            let device = require_device(state, &address)?;
385            let data = std::fs::read(&file).map_err(|e| CliError::AudioReadFailed {
386                path: file.clone(),
387                reason: e.to_string(),
388            })?;
389            device.upload_ringtone(&data, signature.bytes()).await?;
390            println!("Ringtone uploaded: {signature}");
391        }
392        Commands::Monitor { address, duration } => {
393            let device = require_device(state, &address)?;
394            crate::command::monitor::run_monitor(device, duration).await?;
395        }
396    }
397    Ok(())
398}

400/// Get the connected device for the given address, or error.
401fn require_device<'a>(state: &'a ReplState, address: &MacAddress) -> Result<&'a ClockDevice, CliError> {
402    match state.device() {
403        Some(device) if device.address() == address => Ok(device),
404        Some(device) => Err(CliError::Core(ClockError::Parse(format!("connected to {}, but command targets {address}", device.address())))),
405        None => Err(CliError::Core(ClockError::Parse("not connected. Use `connect <mac>` first.".into()))),
406    }
407}

409/// Print REPL help.
410fn print_help() {
411    println!("REPL commands:");
412    println!("  connect <mac>          Connect to a device");
413    println!("  disconnect             Disconnect from current device");
414    println!("  scan [-d <secs>]       Scan for devices");
415    println!("  exit | quit            Exit the REPL");
416    println!();
417    println!("Device commands (require connection):");
418    println!("  sync-time <mac>        Synchronize device clock");
419    println!("  firmware <mac>         Read firmware version");
420    println!("  battery <mac>          Read battery level");
421    println!("  settings-read <mac>    Read device settings");
422    println!("  settings-write <mac> [options]");
423    println!("  alarm-list <mac>       List alarms");
424    println!("  alarm-set <mac> <slot> <HH:MM> [options]");
425    println!("  alarm-delete <mac> <slot>");
426    println!("  brightness <mac> <0-150>");
427    println!("  ringtone-preview <mac> [-v <vol>]");
428    println!("  monitor <mac> [-d <secs>]");
