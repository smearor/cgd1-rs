6/// Arguments for the `alarm-list` command.
7pub struct AlarmListArgs {
8    /// Device MAC address.
9    pub address: MacAddress,
10}
11
12/// Run the `alarm-list` command.
13pub async fn run(args: AlarmListArgs) -> Result<(), CliError> {
14    let connection = DeviceConnection::connect(&args.address).await?;
15    let slots = connection.device().read_alarms().await?;
16
17    if slots.is_empty() {
18        println!("No alarms set.");
19        return Ok(());
20    }
21
22    println!("{:<6} {:<8} {:<3} {:<3} {:<6} {:<6}", "Slot", "Enabled", "HH", "MM", "Days", "Snooze");
23    for slot in &slots {
24        println!(
25            "{:<6} {:<8} {:02}   {:02}   0x{:02x}  {}",
26            slot.index.value(),
27            if slot.entry.enabled() { "yes" } else { "no" },
28            slot.entry.hour(),
29            slot.entry.minute(),
30            slot.entry.repeat_mask().value(),
31            if slot.entry.snooze() { "yes" } else { "no" },
32        );
33    }
34
35    Ok(())
36}
