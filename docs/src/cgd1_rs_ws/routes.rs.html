15/// Build the axum router with all REST and WebSocket routes.
16pub fn build_router(state: ServerState) -> Router {
17    Router::new()
18        .route("/", get(index_page))
19        .route("/health", get(health_check))
20        .route("/api/devices", get(list_devices))
21        .route("/api/devices/{address}/sensors", get(get_sensors))
22        .route("/api/devices/{address}/battery", get(get_battery))
23        .route("/api/devices/{address}/firmware", get(get_firmware))
24        .route("/api/devices/{address}/alarms", get(get_alarms))
25        .route("/api/devices/{address}/settings", get(get_settings))
26        .route("/ws", get(ws_handler))
27        .with_state(state)
28}

30/// Serve the single-file HTML web app.
31async fn index_page() -> impl IntoResponse {
32    Html(include_str!("../static/index.html"))
33}
34
35/// Health check endpoint.
36async fn health_check() -> impl IntoResponse {
37    Json(json!({ "status": "ok" }))
38}
39
40/// List all connected device addresses.
41async fn list_devices(State(state): State<ServerState>) -> impl IntoResponse {
42    let addresses = state.manager().connected_devices().await;
43    Json(json!({ "devices": addresses }))
44}
45
46/// Get latest sensor data for a device.
47async fn get_sensors(State(state): State<ServerState>, Path(address): Path<MacAddress>) -> Result<Json<serde_json::Value>, ServerError> {
48    let device = state.device(&address).await?;
49    let settings = device.read_settings().await?;
50    Ok(Json(serde_json::to_value(&settings)?))
51}

53/// Get battery level for a device.
